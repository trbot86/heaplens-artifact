205 /* NEC LE-IT: compare and swap */
206 /* Returns nonzero if the comparison succeeded. */
207 AO_INLINE int
208 AO_compare_and_swap(volatile AO_t *addr,
209  AO_t old_val, AO_t new_val)
210 {
211  AO_t result,tmp;
212 
213  __asm__ __volatile__("@ AO_compare_and_swap\n"
214 "1: mov %0, #2\n" /* store a flag */
215 " ldrex %1, [%3]\n" /* get original */
216 " teq %1, %4\n" /* see if match */
217 " it eq\n"
218 " strexeq %0, %5, [%3]\n" /* store new one if matched */
219 " teq %0, #1\n"
220 " beq 1b\n" /* if update failed, repeat */
221  : "=&r"(result), "=&r"(tmp), "+m"(*addr)
222  : "r"(addr), "r"(old_val), "r"(new_val)
223  : "cc");
224 
225  return !(result&2); /* if succeded, return 1, else 0 */
226 }

272  /* The first operand is the result, the second the value */
