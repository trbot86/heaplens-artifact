45  template <typename T>
46  void printLogarithmicHistogram(T * valuesOverTime, int numValues) {
47  constexpr int numBits = sizeof(T)*8;
48  int histogram[numBits+1];
49  memset(histogram, 0, sizeof(histogram));
50  T sum = 0;
51  int cntNonZero = 0;
52  for (int i=0;i<numValues;++i) {
53  T v = valuesOverTime[i];
54  if (v == 0) continue;
55  sum += v;
56  ++cntNonZero;
57  int pow2 = 0;
58  while (v > 1) {
59  v >>= 1;
60  ++pow2;
61  }
62  assert(pow2 <= numBits);
63  ++histogram[pow2];
64  }
65  for (int i=0;i<=numBits;++i) {
66  if (histogram[i] > 0) {
67  std::cout<<" (2^"<<twoDigits(i)<<", 2^"<<twoDigits(i+1)<<"]: "<<histogram[i]<<std::endl;
68  }
69  }
70  std::cout<<" average = "<<(sum / (double) cntNonZero)<<std::endl;
71  }

108  inline void DEBUG_RECORD_UPDATE_CHECKSUM(const int tid, const long long timestamp, Node * const * const insertedNodes, Node * const * const deletedNodes, DataStructure * const ds) {
109  #ifdef RQ_VALIDATION

120  for (int j=0;j<cnt;++j) {
121  threadUpdateChecksum[tid][timestamp] += outputKeys[j];

125  K outputKeys[RQ_DEBUGGING_MAX_KEYS_PER_NODE];
126  V outputValues[RQ_DEBUGGING_MAX_KEYS_PER_NODE];
127  int cnt = ds->getKeys(tid, deletedNodes[i], outputKeys, outputValues);
128  assert(cnt <= RQ_DEBUGGING_MAX_KEYS_PER_NODE);
129  for (int j=0;j<cnt;++j) {
130  threadUpdateChecksum[tid][timestamp] -= outputKeys[j];
131  }
132  }
133  #endif
134  }
