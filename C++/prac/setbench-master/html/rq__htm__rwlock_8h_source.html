353  // invoke each time a traversal visits a node with a key in the desired range:
354  // if the node belongs in the range query, it will be placed in rqResult[index]
355  inline int __traversal_try_add(const int tid, NodeType * const node, NodeType ** const nodeSource, K * const outputKeys, V * const outputValues, const K& lo, const K& hi, bool foundDuringTraversal) {
356 
357  // rqResultKeys should have space for MAX_KEYS_PER_NODE keys, AT LEAST
358 
359  // in the following, rather than having deeply nested if-else blocks,
360  // we return asap, and list facts that must be true if we didn't return
361  assert(foundDuringTraversal || !logicalDeletion || ds->isLogicallyDeleted(tid, node));
362 
363  long long itime = TIMESTAMP_NOT_SET;
364  while (itime == TIMESTAMP_NOT_SET) { itime = node->itime; }
365  if (node->itime >= threadData[tid].rq_lin_time) return 0; // node was inserted after the range query
366  // fact: node was inserted before the range query
367 
368  bool logicallyDeleted = (logicalDeletion && ds->isLogicallyDeleted(tid, node));
369  long long dtime = TIMESTAMP_NOT_SET;
370 
371  if (!logicalDeletion && foundDuringTraversal) goto tryAddToRQ; // no logical deletion. since node was inserted before the range query, and the traversal encountered it, it must have been deleted AFTER the traversal encountered it.
372  // fact: no logical deletion ==> did not find node during traversal
373 
374  dtime = node->dtime;
375  if (dtime != TIMESTAMP_NOT_SET) {
376  if (dtime < threadData[tid].rq_lin_time) return 0; // node was deleted before the range query
377  goto tryAddToRQ;
378  }
379 
380  // fact: dtime was not set above
381  if (logicalDeletion && !logicallyDeleted) goto tryAddToRQ; // if logical deletion is used with marking, the fact that node was inserted before the range query, and that the traversal encountered node, is NOT enough to argue that node was in the data structure when the traversal started. why? when the traversal encountered node, it might have already been marked. so, we check if node is marked. if not, then the node has not yet been deleted.
382  // fact: if there is logical deletion, then the node has now been deleted
383 
385  // if we are executing this because node was ANNOUNCED by a process,
386  // as something that MIGHT soon be deleted (if nodeSource != NULL),
387  // then node might not ever actually be deleted,
388  // so we can't spin forever on dtime.
389  if (nodeSource != NULL) {
390  while (dtime == TIMESTAMP_NOT_SET && *nodeSource == node) { dtime = node->dtime; }
391  if (dtime == TIMESTAMP_NOT_SET) {
392  // above loop exited because the process removed its announcement to this node!
393  // if the process deleted the node, then it removed the
394  // announcement AFTER setting dtime.
395  // so we reread dtime one more time, to figure out whether
396  // the process actually deleted the node.
397  SOFTWARE_BARRIER; // prevent read of dtime from happening before last read of *nodeSource
398  dtime = node->dtime;
399  if (dtime == TIMESTAMP_NOT_SET) {
400  // since dtime is not set, the process did NOT delete the node.
401  // so, either a DIFFERENT process deleted it,
402  // or it was found during the data structure traversal.
403  // if another process deleted it, then we will find it
404  // either in that process' announcements, or in a limbo bag.
405  return 0;
406  }
407  // the node has been deleted, and dtime is set, so we check dtime below.
408  }
409  } else {
410  while (dtime == TIMESTAMP_NOT_SET) { dtime = node->dtime; }
411  }
412  if (dtime < threadData[tid].rq_lin_time) return 0; // node was deleted before the range query
413  // fact: node was inserted before the rq and deleted after it
414 
416  // note: this way of organizing this decision tree favors trees with fat multi-key nodes, because getKeys is delayed as long as possible.
417 
418 tryAddToRQ:
419  // fetch the node's keys that are in the set
420  int cnt = ds->getKeys(tid, node, outputKeys, outputValues);
421  assert(cnt < RQ_DEBUGGING_MAX_KEYS_PER_NODE);
422  if (cnt == 0) return 0; // node doesn't contain any keys that are in the set
423  // TODO: properly assert that getKeys doesn't run out of bounds on outputKeys[...] (i'm quite certain it doesn't, currently, though.)
424 
425  // note: in the following loop, we shift keys in the outputKeys array left to eliminate any that ultimately should not be added to the range query
426  int numNewKeys = 0;
427  for (int i=0;i<cnt;++i) { // decide whether key = outputKeys[i] should be in the range query
428  if (!ds->isInRange(outputKeys[i], lo, hi)) goto doNotAddToRQ; // key is NOT in the desired range
429  if (threadData[tid].hashlist->contains(outputKeys[i])) goto doNotAddToRQ; // key is already in the range query
430  outputKeys[numNewKeys] = outputKeys[i]; // save this as a new key added to the RQ
431  outputValues[numNewKeys] = outputValues[i];
432  ++numNewKeys;
433 
434 doNotAddToRQ: (0);
435  }
436  return numNewKeys;
437  }
