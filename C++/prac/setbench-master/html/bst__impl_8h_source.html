447 template<class K, class V, class Compare, class RecManager>
448 int bst_ns::bst<K,V,Compare,RecManager>::rangeQuery(const int tid, const K& lo, const K& hi, K * const resultKeys, V * const resultValues) {
449  block<Node<K,V> > stack (NULL);
450  auto guard = recmgr->getGuard(tid, true);
451  rqProvider->traversal_start(tid);
452 
453  // depth first traversal (of interesting subtrees)
454  int size = 0;
455  stack.push(root);
456  while (!stack.isEmpty()) {
457  Node<K,V> * node = stack.pop();
458  assert(node);
459  Node<K,V> * left = rqProvider->read_addr(tid, &node->left);
460 
461  // if internal node, explore its children
462  if (left != NULL) {
463  if (node->key != this->NO_KEY && !cmp(hi, node->key)) {
464  Node<K,V> * right = rqProvider->read_addr(tid, &node->right);
465  assert(right);
466  stack.push(right);
467  }
468  if (node->key == this->NO_KEY || cmp(lo, node->key)) {
469  assert(left);
470  stack.push(left);
471  }
472 
473  // else if leaf node, check if we should add its key to the traversal
474  } else {
475  rqProvider->traversal_try_add(tid, node, resultKeys, resultValues, &size, lo, hi);
476  }
477  }
478  rqProvider->traversal_end(tid, resultKeys, resultValues, &size, lo, hi);
479  return size;
480 }

483 const std::pair<V,bool> bst_ns::bst<K,V,Compare,RecManager>::find(const int tid, const K& key) {
484  std::pair<V,bool> result;
485  Node<K,V> *p;
486  Node<K,V> *l;
487  for (;;) {
488  TRACE COUTATOMICTID("find(tid="<<tid<<" key="<<key<<")"<<std::endl);
489  auto guard = recmgr->getGuard(tid, true);
490  p = rqProvider->read_addr(tid, &root->left);
491  l = rqProvider->read_addr(tid, &p->left);
492  if (l == NULL) {
493  result = std::pair<V,bool>(NO_VALUE, false); // no keys in data structure
494  return result; // success
495  }
496 
497  while (rqProvider->read_addr(tid, &l->left) != NULL) {
498  TRACE COUTATOMICTID("traversing tree; l="<<*l<<std::endl);
499  p = l; // note: the new p is currently protected
500  assert(p->key != NO_KEY);
501  if (cmp(key, p->key)) {
502  l = rqProvider->read_addr(tid, &p->left);
503  } else {
504  l = rqProvider->read_addr(tid, &p->right);
505  }
506  }
507  if (key == l->key) {
508  result = std::pair<V,bool>(l->value, true);
509  } else {
510  result = std::pair<V,bool>(NO_VALUE, false);
511  }
512  return result; // success
513  }
514  assert(0);
515  return std::pair<V,bool>(NO_VALUE, false);
516 }

867 template <class K, class V, class Compare, class RecManager>
868 int bst_ns::bst<K,V,Compare,RecManager>::help(const int tid, tagptr_t tagptr, SCXRecord<K,V> *snap, bool helpingOther) {
869  TRACE COUTATOMICTID("help "<<tagptrToString(tagptr)<<std::endl);
870  SCXRecord<K,V> *ptr = TAGPTR1_UNPACK_PTR(tagptr);
871 
872  // TODO: make SCX_WRITE_STATE into regular write for the owner of this descriptor
873  // (might not work in general, but here we can prove that allfrozen happens
874  // before state changes, and the sequence number does not change until the
875  // owner is finished with his operation; in general, seems like the last
876  // write done by helpers/owner can be a write, not a CAS when done
877  // by the owner)
878  for (int i=helpingOther; i<snap->c.numberOfNodesToFreeze; ++i) { // freeze sub-tree
879  if (snap->c.scxRecordsSeen[i] == LLX_RETURN_IS_LEAF) {
880  TRACE COUTATOMICTID("nodes["<<i<<"] is a leaf");
881  assert(i > 0); // nodes[0] cannot be a leaf...
882  continue; // do not freeze leaves
883  }
884 
885  uintptr_t exp = (uintptr_t) snap->c.scxRecordsSeen[i];
886  bool successfulCAS = snap->c.nodes[i]->scxRecord.compare_exchange_strong(exp, tagptr); // MEMBAR ON X86/64
887  if (successfulCAS || exp == tagptr) continue; // if node is already frozen for our operation
888 
889  // read mutable allFrozen field of descriptor
890  bool succ;
891  bool allFrozen = DESC1_READ_FIELD(succ, ptr->c.mutables, tagptr, MUTABLES_MASK_ALLFROZEN, MUTABLES_OFFSET_ALLFROZEN);
892  if (!succ) return SCXRecord<K,V>::STATE_ABORTED;
893 
894  int newState = (allFrozen) ? SCXRecord<K,V>::STATE_COMMITTED : SCXRecord<K,V>::STATE_ABORTED;
895  TRACE COUTATOMICTID("help return state "<<newState<<" after failed freezing cas on nodes["<<i<<"]"<<std::endl);
896  MUTABLES1_WRITE_FIELD(ptr->c.mutables, snap->c.mutables, newState, MUTABLES_MASK_STATE, MUTABLES_OFFSET_STATE);
897  return newState;
898  }
899 
900  MUTABLES1_WRITE_BIT(ptr->c.mutables, snap->c.mutables, MUTABLES_MASK_ALLFROZEN);
901  for (int i=1; i<snap->c.numberOfNodesToFreeze; ++i) {
902  if (snap->c.scxRecordsSeen[i] == LLX_RETURN_IS_LEAF) continue; // do not mark leaves
903 #if !defined(BROWN_EXT_BST_LF_COLOCATE_MARKED_BIT)
904  snap->c.nodes[i]->marked.store(true, std::memory_order_relaxed); // finalize all but first node
905 #else
906  snap->c.nodes[i]->scxRecord.fetch_or(0x1, std::memory_order_relaxed);
907  // could this be done more efficiently with bit-test-and-set?
908 #endif
909  }
910 
911  // CAS in the new sub-tree (update CAS)
912 // uintptr_t expected = (uintptr_t) snap->nodes[1];
913  rqProvider->linearize_update_at_cas(tid, snap->c.field, snap->c.nodes[1], snap->c.newNode, snap->c.insertedNodes, snap->c.deletedNodes);
914 
915  // todo: add #ifdef CASing scx record pointers to set an "invalid" bit,
916  // and add to llx a test that determines whether a pointer is invalid.
917  // if so, the record no longer exists and no longer needs help.
918  //
919  // the goal of this is to prevent a possible ABA problem that can occur
920  // if there is VERY long lived data in the tree.
921  // specifically, if a node whose scx record pointer is CAS'd by this scx
922  // was last modified by the owner of this scx, and the owner performed this
923  // modification exactly 2^SEQ_WIDTH (of its own) operations ago,
924  // then an operation running now may confuse a pointer to an old version of
925  // this scx record with the current version (since the sequence #'s match).
926  //
927  // actually, there may be another solution. a problem arises only if a
928  // helper can take an effective action not described by an active operation.
929  // even if a thread erroneously believes an old pointer reflects an active
930  // operation, if it simply helps the active operation, there is no issue.
931  // the issue arises when the helper believes it sees an active operation,
932  // and begins helping by looking at the scx record, but the owner of that
933  // operation is currently in the process of initializing the scx record,
934  // so the helper sees an inconsistent scx record and takes invalid steps.
935  // (recall that a helper can take a snapshot of an scx record whenever
936  // the sequence # it sees in a tagged pointer matches the sequence #
937  // it sees in the scx record.)
938  // the solution here seems to be to ensure scx records can be helped only
939  // when they are consistent.
940 
941  MUTABLES1_WRITE_FIELD(ptr->c.mutables, snap->c.mutables, SCXRecord<K comma1 V>::STATE_COMMITTED, MUTABLES_MASK_STATE, MUTABLES_OFFSET_STATE);
942 
943  TRACE COUTATOMICTID("help return COMMITTED after performing update cas"<<std::endl);
944  return SCXRecord<K,V>::STATE_COMMITTED; // success
945 }
