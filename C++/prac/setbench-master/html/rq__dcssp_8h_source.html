497  // so, we read *snap.addr2. if we see that it contains tagptr, then we can help the dcssp.
498  // otherwise, one of the following must be true:
499  // (a) otherTid has not yet changed *snap.addr2 to tagptr, or
500  // (b) otherTid changed *snap.addr2 to tagptr, then it (or a helper) changed its state to SUCCEEDED, then changed *snap.addr2 to a different value (never again to contain tagptr), or
501  // (c) otherTid changed *snap.addr2 to tagptr, then it (or a helper) changed its state to FAILED, then changed *snap.addr2 to a different value (never again to contain tagptr).
502  // in case (a), we know that the dcssp has not yet been linearized.
503  // in case (b), the dcssp has been linearized, has state SUCCEEDED, and deleted node.
504  // in case (c), the dcssp has been linearized, has state FAILED, and did NOT delete node.
505  // so, after reading *snap.addr2, we read the dcssp operation's state again to determine which case has occurred.
506

508  if (state == DCSSP_STATE_FAILED) { // the operation described by snap did not insert/delete node, so either this process did not insert/delete it, or the process inserted/deleted it in a PREVIOUS operation, so it must have already set itime/dtime as appropriate
509  continue; // goto check next process
510  } else if (state == DCSSP_STATE_SUCCEEDED) { // the dcssp operation finished, and deleted node. to determine WHEN it deleted node, we look at the argument old1 to the dcssp, which contains the timestamp when the dcssp took place. (observe that this is the value process otherTid would write to node->dtime)
511  if (WAIT_FOR_DTIME(node)) {
512  // the following assertions are thread safe ONLY if WAIT_FOR_DTIME actually waits! (which is true only if it returns true, which is true only if RQ_LOCKFREE_WAITS_FOR_DTIME is defined)
513  assert(snap.old1 <= node->dtime);
514  assert((snap.old1 >= threadData[tid].rq_lin_time) == (node->dtime >= threadData[tid].rq_lin_time));
515  assert(foundDuringTraversal || node->dtime == snap.old1);
516  assert(!foundDuringTraversal || node->dtime == snap.old1);
517  }
518  if (snap.old1 < threadData[tid].rq_lin_time) return 0; // node was deleted before rq
519  goto tryAddToRQ; // node was deleted by this process after rq
520  }
521  // fact: state is UNDECIDED
522 
523  // maya: in logicalDeletion, since the node is marked the DCSS was successful and only the dtime is not yet set. Thus, UNDECIDED means that otherThread did not mark the node, it was some other thread and we can continue.
524  if (logicalDeletion) continue; // goto check next process
525 
526  // now we try to help
527  casword_t addr2 = *snap.addr2;
528  if (addr2 == tagptr) { // TODO: prove it is impossible to execute this block with logical deletion (idea: since node is marked, either (1) otherTid marked it earlier with a DCSS whose state is SUCCEEDED, or (2) someone else marked node, so otherTid cannot successfully CAS addr2.)
529  // addr2 indeed points to the dcssp descriptor. the linearization point of the dcssp operation occurs after this step, so the dcssp might have been linearized, but not yet had its state set.
530  prov->helpProcess(tid, otherTid); // we need to know what its final state will be to determine whether it successfully deleted node. so, we HELP otherTid finish its dcssp.
531  }
532  // note: the following all happens in BOTH the cases where addr2 == tagptr and where addr2 != tagptr, except there is some extra work if addr2 != tagptr and state2 != SUCCEEDED. i've folded the two cases together simply for compactness / less repetition.
533 
534  // then, we reread the state
535  bool valid = false;
536  dcsspdesc_t<void *> * ptr = prov->getDescriptorPtr(tagptr);
537  int state2 = DESC_READ_FIELD(valid, ptr->mutables, tagptr, DCSSP_MUTABLES_MASK_STATE, DCSSP_MUTABLES_OFFSET_STATE);
538  if (!valid) continue; // goto check next process // the read of the state field was invalid, which means that the dcssp operation has terminated, the next dcssp operation by otherTid has begun. since the next dcssp operation has begun, and each high-level data structure operation performs only one successful dcssp (in a call to linearize_update_at_...), if this dcssp that finished in fact deleted node, then the next dcssp would be part of the next high-level operation. thus, if node->itime was inserted by the finished dcssp, then the high-level operation that performed this dcssp will already have set node->itime.
539  // fact: the read of state was valid
540 
541  if (state2 == DCSSP_STATE_SUCCEEDED) { // we are in case (b) described above. the dcssp operation finished, and deleted node. to determine WHEN it deleted node, we look at the argument old1 to the dcssp, which contains the timestamp when the dcssp took place. (observe that this is the value process otherTid would write to node->dtime)
542  if (snap.old1 >= threadData[tid].rq_lin_time) goto tryAddToRQ;
543  return 0; // do not add to rq
544  } else { // undecided or failed
545  continue; // goto check next process // we are in case (a) or case (c) described above. so, otherTid did NOT delete node.
546  }
547  }
548  if(dtime == TIMESTAMP_NOT_SET) {
549  assert(!logicalDeletion);
550  assert(!foundDuringTraversal);
551  goto tryAddToRQ; // no process deleted node before the range query
552  }
553  COUNT_CODE_PATH(9);
554  if (dtime >= threadData[tid].rq_lin_time) goto tryAddToRQ;
555  return 0; // do not add to rq
556 
558  // note: this way of organizing this decision tree favors trees with fat multi-key nodes, because getKeys is delayed as long as possible.
559 
560 tryAddToRQ:
561  // fetch the node's keys that are in the set
562  int cnt = ds->getKeys(tid, node, outputKeys, outputValues);
563  assert(cnt < RQ_DEBUGGING_MAX_KEYS_PER_NODE);
564  if (cnt == 0) return 0; // node doesn't contain any keys that are in the set and in the desired range

567  int numNewKeys = 0;

571  outputKeys[numNewKeys] = outputKeys[i]; // save this as a new key added to the RQ
572  outputValues[numNewKeys] = outputValues[i];
573  ++numNewKeys;
574 
575 doNotAddToRQ: (0);
576  }
577  return numNewKeys;
