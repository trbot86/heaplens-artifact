243 template <typename skey_t, typename sval_t, class RecMgr>
244 void intlf<skey_t, sval_t, RecMgr>::seek(tArgs<skey_t, sval_t>* t, skey_t key, seekRecord<skey_t, sval_t>* s) {
245  anchorRecord<skey_t, sval_t>* pAnchorRecord;
246  anchorRecord<skey_t, sval_t>* anchorRecord;
247 
248  struct edge<skey_t, sval_t> pLastEdge;
249  struct edge<skey_t, sval_t> lastEdge;
250 
251  node_t<skey_t, sval_t> * curr;
252  node_t<skey_t, sval_t> * next;
253  node_t<skey_t, sval_t> * temp;
254  int which;
255 
256  bool n;
257  bool d;
258  bool p;
259  skey_t cKey;
260  skey_t aKey;
261 
262  pAnchorRecord = &t->pAnchorRecord;
263  anchorRecord = &t->anchorRecord;
264 
265  pAnchorRecord->node = S;
266  pAnchorRecord->key = KEY_MAX;
267 
268  while (true) {
269  //initialize all variables used in traversal
270  populateEdge(&pLastEdge, R, S, RIGHT);
271  populateEdge(&lastEdge, S, T, RIGHT);
272  curr = T;
273  anchorRecord->node = S;
274  anchorRecord->key = KEY_MAX;
275  while (true) {
276  //read the key stored in the current node
277  cKey = getKey(curr->markAndKey);
278  //find the next edge to follow
279  which = key < cKey ? LEFT : RIGHT;
280  temp = curr->child[which];
281  n = isNull(temp);
282  d = isDFlagSet(temp);
283  p = isPFlagSet(temp);
284  next = getAddress(temp);
285  //std::cout<<"ckey="<<cKey<<" search key="<<key<<" which="<<which<<" n="<<n<<" d="<<d<<" p="<<p<<" next="<<next<<std::endl;
286  //check for completion of the traversal
287  if (key == cKey || n) {
288  //either key found or no next edge to follow. Stop the traversal
289  s->pLastEdge = pLastEdge;
290  s->lastEdge = lastEdge;
291  populateEdge(&s->injectionEdge, curr, next, which);
292  if (key == cKey) {
293  //key matches. So return
294  return;
295  } else {
296  break;
297  }
298  }
299  if (which == RIGHT) {
300  //the next edge that will be traversed is a right edge. Keep track of the current node and its key
301  anchorRecord->node = curr;
302  anchorRecord->key = cKey;
303  }
304  //traverse the next edge
305  pLastEdge = lastEdge;
306  populateEdge(&lastEdge, curr, next, which);
307  curr = next;
308  }
309  //key was not found. check if can stop
310  temp = anchorRecord->node->child[RIGHT];
311  d = isDFlagSet(temp);
312  p = isPFlagSet(temp);
313  if (!d && !p) {
314  //the anchor node is part of the tree. Return the results of the current traversal
315  aKey = getKey(anchorRecord->node->markAndKey);
316  if (anchorRecord->key == aKey) return;
317  } else {
318  if (pAnchorRecord->node == anchorRecord->node && pAnchorRecord->key == anchorRecord->key) {
319  //return the results of previous traversal
320  copySeekRecord(s, &t->pSeekRecord);
321  return;
322  }
323  }
324  //store the results of the current traversal and restart
325  copySeekRecord(&t->pSeekRecord, s);
326  pAnchorRecord->node = anchorRecord->node;
327  pAnchorRecord->key = anchorRecord->key;
328  }
329 }

507 void intlf<skey_t, sval_t, RecMgr>::findAndMarkSuccessor(tArgs<skey_t, sval_t>* t, stateRecord<skey_t, sval_t>* state) {

510  bool n;
511  bool d;
512  bool p;

514  node_t<skey_t, sval_t> * left;
515 
516  //retrieve the addresses from the state record
517  auto node = state->targetEdge.child;
518  auto s = &state->successorRecord;
519  while (true) {
520  //read the mark flag of the key in the target node
521  m = isKeyMarked(node->markAndKey);
522  //find the node with the smallest key in the right subtree
523  findSmallest(t, node, s);
524  if (m) {
525  //successor node has already been selected before the traversal
526  break;
527  }
528  //retrieve the information from the seek record
529  successorEdge = s->lastEdge;
530  temp = successorEdge.child->child[LEFT];
531  n = isNull(temp);
532  p = isPFlagSet(temp);
533  left = getAddress(temp);
534  if (!n) {
535  continue;
536  }
537  //read the mark flag of the key under deletion
538  m = isKeyMarked(node->markAndKey);
539  if (m) {
540  //successor node has already been selected
541  if (p) {
542  break;
543  } else {
544  continue;
545  }
546  }
547  //try to set the promote flag on the left edge
548  if (CAS(successorEdge.child, LEFT, setNull(left), setPFlag(setNull(node)))) break;
549  //attempt to mark the edge failed; recover from the failure and retry if needed
550  temp = successorEdge.child->child[LEFT];
551  n = isNull(temp);
552  d = isDFlagSet(temp);
553  p = isPFlagSet(temp);
554  if (p) break;

644  } else {

652  }

657  } else {
658  oldValue = successorEdge.child;
659  newValue = right;
660  }
661  }
