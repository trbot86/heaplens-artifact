219  node_t<skey_t, sval_t> * n = root;
220  node_t<skey_t, sval_t> * child;
221  skey_t curr_key;
222  while (1) {
223  curr_key = n->key;
224  if (curr_key == k) {
225  return n;
226  }
227  if (curr_key < k) {
228  child = n->right;
229  } else {
230  child = n->left;
231  }
232  if (child == NULL) {
233  return n;
234  }
235  n = child;

272 
273  if (onlyIfAbsent) {
274  auto n = node;
275  while (n->key > k) {
276  n = n->pred;
277  }
278  while (n->key < k) {
279  n = n->succ;
280  }
281  if ((n->key == k) && (n->mark == false)) {
282  return n->value;
283  }

286  LOCK(&(p->succ_lock));
287 #if defined(NO_VOLATILE) || defined(BASELINE)

289 #else
