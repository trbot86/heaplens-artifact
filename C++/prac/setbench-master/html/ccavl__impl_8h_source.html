675 template <typename skey_t, typename sval_t, class RecMgr>
676 sval_t ccavl<skey_t, sval_t, RecMgr>::attemptUpdate(
677  const int tid,
678  skey_t key,
679  int func,
680  sval_t expected,
681  sval_t newValue,
682  node_t<skey_t, sval_t>* parent,
683  node_t<skey_t, sval_t>* curr,
684  version_t nodeOVL) {
685  // As the search progresses there is an implicit min and max assumed for the
686  // branch of the tree rooted at node. A left rotation of a node x results in
687  // the range of keys in the right branch of x being reduced, so if we are at a
688  // node and we wish to traverse to one of the branches we must make sure that
689  // the node has not undergone a rotation since arriving from the parent.
690  //
691  // A rotation of node can't screw us up once we have traversed to node's
692  // child, so we don't need to build a huge transaction, just a chain of
693  // smaller read-only transactions.
694  //long cmp;
695  char dirToC;
696 
697  assert(parent != curr);
698  assert(nodeOVL != UnlinkedOVL);
699 
700  //cmp = key - curr->key;
701  if (key == curr->key) {
702  return attemptNodeUpdate(tid, func, expected, newValue, parent, curr);
703  }
704 
705  dirToC = key < curr->key ? LEFT : RIGHT;
706 
707  while (1) {
708  node_t<skey_t, sval_t>* child = get_child(curr, dirToC);
709 
710  if (hasShrunkOrUnlinked(nodeOVL, curr->changeOVL)) {
711  return (sval_t) SpecialRetry;
712  }
713 
714  if (child == NULL) {
715  // key is not present
716  if (newValue == NULL) {
717  // Removal is requested. Read of node.child occurred
718  // while parent.child was valid, so we were not affected
719  // by any shrinks.
720  return NULL;
721  } else {
722  // Update will be an insert.
723  int success;
724  node_t<skey_t, sval_t>* damaged;
725  mutex_lock(&(curr->lock));
726  {
727  // Validate that we haven't been affected by past
728  // rotations. We've got the lock on node, so no future
729  // rotations can mess with us.
730  if (hasShrunkOrUnlinked(nodeOVL, curr->changeOVL)) {
731  mutex_unlock(&(curr->lock));
732  return (sval_t) SpecialRetry;
733  }
734 
735  if (get_child(curr, dirToC) != NULL) {
736  // Lost a race with a concurrent insert. No need
737  // to back up to the parent, but we must RETRY in
738  // the outer loop of this method.
739  success = 0;
740  damaged = NULL;
741  } else {
742  // We're valid. Does the user still want to
743  // perform the operation?
744  if (!shouldUpdate(func, NULL, expected)) {
745  mutex_unlock(&(curr->lock));
746  return NULL;
747  }
748 
749  // Create a new leaf
750  setChild(curr, dirToC, rbnode_create(tid, key, newValue, curr));
751  success = 1;
752 
753  // attempt to fix node.height while we've still got
754  // the lock
755  damaged = fixHeight_nl(curr);
756  }
757  }
758  mutex_unlock(&(curr->lock));
759  if (success) {
760  fixHeightAndRebalance(tid, damaged);
761  return NULL;
762  }
763  // else RETRY
764  }
765  } else {
766  // non-null child
767  version_t childOVL = child->changeOVL;
768  if (isShrinkingOrUnlinked(childOVL)) {
769  waitUntilChangeCompleted(child, childOVL);
770  // RETRY
771  } else if (child != get_child(curr, dirToC)) {
772  // this second read is important, because it is protected
773  // by childOVL
774  // RETRY
775  } else {
776  // validate the read that our caller took to get to node
777  if (hasShrunkOrUnlinked(nodeOVL, curr->changeOVL)) {
778  return (sval_t) SpecialRetry;
779  }
780 
781  // At this point we know that the traversal our parent took
782  // to get to node is still valid. The recursive
783  // implementation will validate the traversal from node to
784  // child, so just prior to the nodeOVL validation both
785  // traversals were definitely okay. This means that we are
786  // no longer vulnerable to node shrinks, and we don't need
787  // to validate nodeOVL any more.
788  sval_t vo = attemptUpdate(tid, key, func,
789  expected, newValue, curr, child, childOVL);
790  if (vo != (sval_t) SpecialRetry) {
791  return vo;
792  }
793  // else RETRY
794  }
795  }
796  }
797 }

844  if (newValue == NULL) {
845  // removal
846  if (curr->value == NULL) {
847  // This node is already removed, nothing to do.
848  return NULL;
849  }
850  }
851 
852  if (newValue == NULL && (curr->left == NULL || curr->right == NULL)) {
853  // potential unlink, get ready by locking the parent

886  // potential update (including remove-without-unlink)

888  {

890  if (isUnlinked(curr->changeOVL)) {
891  mutex_unlock(&(curr->lock));
892  return (sval_t) SpecialRetry;
893  }
894 
895  prev = curr->value;
896  if (!shouldUpdate(func, prev, expected)) {
897  mutex_unlock(&(curr->lock));
898  return prev;
899  }
900 
901  // retry if we now detect that unlink is possible
902  if (newValue == NULL && (curr->left == NULL || curr->right == NULL)) {
903  mutex_unlock(&(curr->lock));
904  return (sval_t) SpecialRetry;
905  }
906 
907  // update in-place
908  curr->value = newValue;
909  mutex_unlock(&(curr->lock));
910  return prev;
911  }
912  mutex_unlock(&(curr->lock));
913  }

982 int ccavl<skey_t, sval_t, RecMgr>::nodeCondition(node_t<skey_t, sval_t>* curr) {
983  // Begin atomic.
984

987  int hR0;
988  int hNRepl;
989  int bal;
990  node_t<skey_t, sval_t>* nL = (node_t<skey_t, sval_t>*) curr->left;
991  node_t<skey_t, sval_t>* nR = (node_t<skey_t, sval_t>*) curr->right;

1007  bal = hL0 - hR0;

1129 template <typename skey_t, typename sval_t, class RecMgr>
1130 node_t<skey_t, sval_t>* ccavl<skey_t, sval_t, RecMgr>::rebalanceToRight_nl(node_t<skey_t, sval_t>* nParent, node_t<skey_t, sval_t>* n,
1131  node_t<skey_t, sval_t>* nL, int hR0) {
1132  node_t<skey_t, sval_t>* result;
1133 
1134  // L is too large, we will rotate-right. If L.R is taller
1135  // than L.L, then we will first rotate-left L.
1136  {
1137  int hL = nL->height;
1138  if (hL - hR0 <= 1) {
1139  return n; // retry
1140  } else {
1141  node_t<skey_t, sval_t>* nLR = (node_t<skey_t, sval_t>*) nL->right;
1142  int hLL0 = height((node_t<skey_t, sval_t>*) nL->left);
1143  int hLR0 = height(nLR);
1144  if (hLL0 >= hLR0) {
1145  // rotate right based on our snapshot of hLR
1146  if (nLR != NULL) mutex_lock(&(nLR->lock));
1147  result = rotateRight_nl(nParent, n, nL, nLR, hR0, hLL0, hLR0);
1148  if (nLR != NULL) mutex_unlock(&(nLR->lock));
1149  return result;
1150  } else {
1151  mutex_lock(&(nLR->lock));
1152  {
1153  // If our hLR snapshot is incorrect then we might
1154  // actually need to do a single rotate-right on n.
1155  int hLR = nLR->height;
1156  if (hLL0 >= hLR) {
1157  result = rotateRight_nl(nParent, n, nL, nLR, hR0, hLL0, hLR);
1158  mutex_unlock(&(nLR->lock));
1159  return result;
1160  } else {
1161  // If the underlying left balance would not be
1162  // sufficient to actually fix n.left, then instead
1163  // of rolling it into a double rotation we do it on
1164  // it's own. This may let us avoid rotating n at
1165  // all, but more importantly it avoids the creation
1166  // of damaged nodes that don't have a direct
1167  // ancestry relationship. The recursive call to
1168  // rebalanceToRight_nl in this case occurs after we
