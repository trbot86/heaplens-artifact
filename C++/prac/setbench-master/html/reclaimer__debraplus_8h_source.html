192  // rotate the epoch bags and reclaim any objects retired two epochs ago.
193  inline void rotateEpochBags(const int tid) {
194  assert(isQuiescent(tid));
195  // we rotate lists in constant time, and scan hazard pointers
196  // when the blockbag from two epochs ago is larger than scanThreshold
197  // (using an iterator with erase functionality).
198  // maybe in the future we could use bloom filters somehow to determine when no hazard pointer
199  // can be present in a block, so we can reclaim the entire block in O(k) time...???
200  // (if we're willing to accept k full, unreclaimable blocks per thread, then we can avoid
201  // working with individual elements altogether. we can simply check if each HP is in the bloom
202  // filter for each of c*n blocks (for some constant c), and have some probability of being
203  // able to reclaim (c-1)*n blocks. then, this procedure will be worst case O(n) time.)
204 
205  index[tid*PREFETCH_SIZE_WORDS] = (index[tid*PREFETCH_SIZE_WORDS]+1) % NUMBER_OF_EPOCH_BAGS_CR;
206  blockbag<T> * const freeable = epochbags[NUMBER_OF_EPOCH_BAGS_CR*tid+index[tid*PREFETCH_SIZE_WORDS]];
207  if (freeable->getSizeInBlocks() >= scanThreshold) {
208 
209  TRACE COUTATOMICTID("retiring... we have "<<freeable->computeSize()<<" things waiting to be retired in this epoch bag...");
210  // hash all announcements
211  comparing[tid]->clear();
212  assert(comparing[tid]->size() == 0);
213  for (int otherTid=0; otherTid < this->NUM_PROCESSES; ++otherTid) {
214  int sz = announce[otherTid]->size();
215  for (int ixHP = 0; ixHP < sz; ++ixHP) {
216  T* hp = (T*) announce[otherTid]->get(ixHP);
217  if (hp) {
218  int oldSize; DEBUG2 oldSize = comparing[tid]->size();
219  comparing[tid]->insert((T*) hp);
220  DEBUG2 assert(comparing[tid]->size() <= oldSize + 1); // might not increase size if comparing[tid] already contains this item...
221  }
222  }
223  }

226  // and swap them to the front of the blockbag.
227  // once all announced nodes are at the front of the blockbag,
228  // we can free whole blocks in the remainder of the blockbag.

249  // invoke this at the beginning of each operation that accesses
250  // objects reclaimed by this epoch manager.
251  // returns true if the call rotated the epoch bags for thread tid
252  // (and reclaimed any objects retired two epochs ago).
253  // otherwise, the call returns false.

255  inline bool startOp(const int tid, void * const * const reclaimers, const int numReclaimers) {
256  SOFTWARE_BARRIER; // prevent any bookkeeping from being moved after this point by the compiler.
257  bool result = false;
258  long readEpoch = epoch; // multiple of EPOCH_INCREMENT
259  assert(!QUIESCENT(readEpoch));
260  // if our announced epoch is different from the current epoch
261  const long ann = announcedEpoch[tid*PREFETCH_SIZE_WORDS].load(std::memory_order_relaxed);
262  DEBUG2 if (!QUIESCENT(ann)) {
263  COUTATOMICTID("NOT QUIESCENT"<<std::endl);
264  exit(-1);
265  }
266  if (readEpoch != BITS_EPOCH(ann)) {
267  // announce the new epoch, and rotate the epoch bags and
268  // reclaim any objects retired two epochs ago.
269  checked[tid*PREFETCH_SIZE_WORDS] = 0;
270  //rotateEpochBags(tid);
271  for (int i=0;i<numReclaimers;++i) {
272  ((reclaimer_debraplus<T, Pool> * const) reclaimers[i])->rotateEpochBags(tid);
273  }
274  result = true;
275  }
276  // note: readEpoch, when written to announcedEpoch[tid],
277  // will set the state to non-quiescent and non-neutralized
278 
279  // incrementally scan the announced epochs of all threads
280  int otherTid = checked[tid*PREFETCH_SIZE_WORDS];
281  if (otherTid >= this->NUM_PROCESSES) {
282  const int c = ++checked[tid*PREFETCH_SIZE_WORDS];
283  if (c > MINIMUM_OPERATIONS_BEFORE_NEW_EPOCH_CR) {
284  __sync_bool_compare_and_swap(&epoch, readEpoch, readEpoch+EPOCH_INCREMENT);
285  }
286  } else {
287  assert(otherTid >= 0);
288  long otherAnnounce = announcedEpoch[otherTid*PREFETCH_SIZE_WORDS].load(std::memory_order_relaxed);
289  if (BITS_EPOCH(otherAnnounce) == readEpoch
290  || QUIESCENT(otherAnnounce)
291  || neutralizeOther(tid, otherTid, readEpoch, otherAnnounce)) {
292  const int c = ++checked[tid*PREFETCH_SIZE_WORDS];
293  if (c >= this->NUM_PROCESSES && c > MINIMUM_OPERATIONS_BEFORE_NEW_EPOCH_CR) {
294  __sync_bool_compare_and_swap(&epoch, readEpoch, readEpoch+EPOCH_INCREMENT);
295  }
296  }
297  }
298  // it is important that we set the announcedEpoch last, because we must
299  // not be neutralized during some of the preceding steps, or we may
300  // corrupt the data structure.
301  // (on x86/64, writes are not moved earlier in program order, so we don't need any membar before this write.)
302  // (on another arch, we'd have to prevent this write from being moved before the write to checked[].)
303  assert(isQuiescent(tid));
304  SOFTWARE_BARRIER;
305  announcedEpoch[tid*PREFETCH_SIZE_WORDS].store(readEpoch, std::memory_order_relaxed);
306  return result;
307  }
