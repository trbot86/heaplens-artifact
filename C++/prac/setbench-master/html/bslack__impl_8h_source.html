950 int bslack_ns::bslack<DEGREE,K,Compare,RecManager>::rangeQuery(const int tid, const K& lo, const K& hi, K * const resultKeys, void ** const resultValues) {
951  block<Node<DEGREE,K>> stack (NULL);
952  auto guard = recordmgr->getGuard(tid, true);
953  rqProvider->traversal_start(tid);
954 
955  // depth first traversal (of interesting subtrees)
956  int size = 0;
957  TRACE COUTATOMICTID("rangeQuery(lo="<<lo<<", hi="<<hi<<", size="<<(hi-lo+1)<<")"<<std::endl);
958 
959  stack.push(entry);
960  while (!stack.isEmpty()) {
961  Node<DEGREE,K> * node = stack.pop();
962  assert(node);
963 
964  // if leaf node, check if we should add its keys to the traversal
965  if (node->isLeaf()) {
966  rqProvider->traversal_try_add(tid, node, resultKeys, resultValues, &size, lo, hi);
967 
968  // else if internal node, explore its children
969  } else {
970  // find right-most sub-tree that could contain a key in [lo, hi]
971  int nkeys = node->getKeyCount();
972  int r = nkeys;
973  while (r > 0 && cmp(hi, (const K&) node->keys[r-1])) --r; // subtree rooted at node->ptrs[r] contains only keys > hi
974 
975  // find left-most sub-tree that could contain a key in [lo, hi]
976  int l = 0;
977  while (l < nkeys && !cmp(lo, (const K&) node->keys[l])) ++l; // subtree rooted at node->ptrs[l] contains only keys < lo
978 
979  // perform DFS from left to right (so push onto stack from right to left)
980  for (int i=r;i>=l; --i) stack.push(rqProvider->read_addr(tid, &node->ptrs[i]));
981 
982 // // simply explore EVERYTHING
983 // for (int i=0;i<node->getABDegree();++i) {
984 // stack.push(rqProvider->read_addr(tid, &node->ptrs[i]));
985 // }
986  }
987  }
988 
989  // success
990  rqProvider->traversal_end(tid, resultKeys, resultValues, &size, lo, hi);
991  return size;
992 }

996 void* bslack_ns::bslack<DEGREE,K,Compare,RecManager>::doInsert(const int tid, const K& key, void * const value, const bool replace) {
997  wrapper_info<DEGREE,K> _info;
998  wrapper_info<DEGREE,K>* info = &_info;
999  while (true) {
1003  auto guard = recordmgr->getGuard(tid);
1004  Node<DEGREE,K>* gp = NULL;
1005  Node<DEGREE,K>* p = entry;
1006  Node<DEGREE,K>* l = rqProvider->read_addr(tid, &p->ptrs[0]);
1007  int ixToP = -1;
1008  int ixToL = 0;
1009  while (!l->isLeaf()) {
1010  ixToP = ixToL;
1011  ixToL = l->getChildIndex(key, cmp);
1012  gp = p;
1013  p = l;
1014  l = rqProvider->read_addr(tid, &l->ptrs[ixToL]);
1015  }
1016 
1020  int keyIndex = l->getKeyIndex(key, cmp);
1021  if (keyIndex < l->getKeyCount() && l->keys[keyIndex] == key) {
1025  void* const oldValue = l->ptrs[keyIndex]; // this is a value, not a pointer, so it cannot be modified by rqProvider->linearize_update_at_..., so we do not use read_addr
1026  if (!replace) {
1027  return oldValue;
1028  }
1029 
1030  // perform LLXs
1031  if (!llx(tid, p, NULL, 0, info->scxPtrs, info->nodes)
1032  || rqProvider->read_addr(tid, &p->ptrs[ixToL]) != l) {
1033  continue; // retry the search
1034  }
1035  info->nodes[1] = l;
1036 
1037  // create new node(s)
1038  Node<DEGREE,K>* n = allocateNode(tid);
1039  arraycopy(l->keys, 0, n->keys, 0, l->getKeyCount());
1040  arraycopy(l->ptrs, 0, n->ptrs, 0, l->getABDegree()); // although we are copying l->ptrs, since l is a leaf, l->ptrs CANNOT contain modified by rqProvider->linearize_update_at_..., so we do not use arraycopy_ptrs.
1041  n->ptrs[keyIndex] = (Node<DEGREE,K>*) value; // similarly, we don't use write_addr here
1042  n->leaf = true;
1043  n->marked = false;
1044  n->scxPtr = DUMMY;
1045  n->searchKey = l->searchKey;
1046  n->size = l->size;
1047  n->weight = true;
1048 
1049  // construct info record to pass to SCX
1050  info->numberOfNodes = 2;
1051  info->numberOfNodesAllocated = 1;
1052  info->numberOfNodesToFreeze = 1;
1053  info->field = &p->ptrs[ixToL];
1054  info->newNode = n;
1055  info->insertedNodes[0] = n;
1056  info->insertedNodes[1] = NULL;
1057  info->deletedNodes[0] = l;
1058  info->deletedNodes[1] = NULL;
1059 
1060  if (scx(tid, info)) {
1061  TRACE COUTATOMICTID("replace std::pair ("<<key<<", "<<value<<"): SCX succeeded"<<std::endl);
1062 #ifndef REBALANCING_NONE
1063  #ifdef USE_SIMPLIFIED_ABTREE_REBALANCING
1064  fixDegreeOrSlackViolation(tid, n);
1065  #endif
1066 #endif
1067  return oldValue;
1068  }

1070  this->recordmgr->deallocate(tid, n);
1071 
1072  } else {
1077  // perform LLXs
1078  if (!llx(tid, p, NULL, 0, info->scxPtrs, info->nodes) || rqProvider->read_addr(tid, &p->ptrs[ixToL]) != l) {
1079  continue; // retry the search
1080  }

1088  // create new node(s)
1089  Node<DEGREE,K>* n = allocateNode(tid);

1095  n->ptrs[keyIndex] = (Node<DEGREE,K>*) value; // similarly, we don't use write_addr here
1096  n->leaf = l->leaf;
1097  n->marked = false;
1098  n->scxPtr = DUMMY;
1099  n->searchKey = l->searchKey;
1100  n->size = l->size+1;
1101  n->weight = l->weight;
1102 
1103  // construct info record to pass to SCX
1104  info->numberOfNodes = 2;
1105  info->numberOfNodesAllocated = 1;
1106  info->numberOfNodesToFreeze = 1;
1107  info->field = &p->ptrs[ixToL];
1108  info->newNode = n;
1109  info->insertedNodes[0] = n;
1110  info->insertedNodes[1] = NULL;
1111  info->deletedNodes[0] = l;
1112  info->deletedNodes[1] = NULL;

1125 
1126  } else { // assert: l->getKeyCount() == DEGREE == b)
1131  // first, we create a pair of large arrays
1132  // containing too many keys and pointers to fit in a single node
1133  K keys[DEGREE+1];
