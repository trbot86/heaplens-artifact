62  // place bindings for the token at the end of customBinding, and
63  // return the index of the first character in the next token,
64  // or the size of the string argv if there are no further tokens.
65  static unsigned parseToken(std::string argv, int ix) {
66  // token is of one of following forms:
67  // INT
68  // INT-INT
69  // and is either followed by "," is the end of the string.
70  // we first determine which is the case
71 
72  // read first INT
73  int ix2 = ix;
74  while (ix2 < argv.size() && argv[ix2] != '.') ++ix2;
75  string token = argv.substr(ix, ix2-ix+1);
76  int a = atoi(token.c_str());
77 
78  // check if the token is of the first form: INT
79  ix = ix+digits(a); // first character AFTER first INT
80  if (ix >= argv.size() || argv[ix] == '.') {
81 
82  // add single binding
83  //cout<<"a="<<a<<std::endl;
84  //customBinding.push_back(a);
85  customBinding[numCustomBindings++] = a;
86 
87  // token is of the second form: INT-INT
88  } else {
89  assert(argv[ix] == '-');
90  ++ix; // skip '-'
91 
92  // read second INT
93  token = argv.substr(ix, ix2-ix+1);
94  int b = atoi(token.c_str());
95  //cout<<"a="<<a<<" b="<<b<<std::endl;
96 
97  // add range of bindings
98  for (int i=a;i<=b;++i) {
99  //customBinding.push_back(i);
100  customBinding[numCustomBindings++] = i;
101  }
102 
103  ix = ix+digits(b); // first character AFTER second INT
104  }
105  // note: ix is the first character AFTER the last INT in the token
106  // this is either a comma ('.') or the end of the string argv.
107  return (ix >= argv.size() ? argv.size() : ix+1 /* skip '.' */);
108  }
