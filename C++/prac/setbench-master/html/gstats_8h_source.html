483  template <typename T>
484  stat_metrics<T> * compute_stat_metrics_total(const gstats_stat_id id) {
485  stat_metrics<T> * results = new stat_metrics<T>[1];
486  acquireLock(&arrays_to_delete_lock);
487  arrays_to_delete.push_back((stat_metrics<double> *) results);
488  releaseLock(&arrays_to_delete_lock);
489 
490  // first, cnt, min, max, sum
491  results[0].first = (NUM_PROCESSES <= 0 || thread_data[0].size[id] <= 0) ? 0 : thread_data[0].get_ptr<T>(id)[0];
492  for (int tid=0;tid<NUM_PROCESSES;++tid) {
493  int size = thread_data[tid].size[id];
494  auto data = thread_data[tid].get_ptr<T>(id);
495  for (int ix=0;ix<size;++ix) {
496  if (data[ix] == 0) continue;
497  ++results[0].cnt;
498  results[0].min = std::min(results[0].min, data[ix]);
499  results[0].max = std::max(results[0].max, data[ix]);
500  results[0].sum += data[ix];
501  }
502  }
503 
504  // avg
505  if (results[0].cnt > 0) results[0].avg = results[0].sum / results[0].cnt;
506 
507  // variance
508  for (int tid=0;tid<NUM_PROCESSES;++tid) {
509  int size = thread_data[tid].size[id];
510  auto data = thread_data[tid].get_ptr<T>(id);
511  for (int ix=0;ix<size;++ix) {
512  if (data[ix] == 0) continue;
513  results[0].variance += GSTATS_SQ(data[ix] - results[0].avg);
514  }
515  }
516 
517  // unbiased sample stdev
518  if (results[0].cnt > 1) results[0].stdev = sqrt(results[0].variance) / (results[0].cnt - 1);
519 
520  return results;
521  }

545  if (data[ix] == 0) continue;
546  ++results[ix].cnt;
547  results[ix].min = std::min(results[ix].min, data[ix]);
548  results[ix].max = std::max(results[ix].max, data[ix]);
549  results[ix].sum += data[ix];
550  }
551  if (results[ix].cnt > 0) results[ix].avg = results[ix].sum / results[ix].cnt;
552  }
553 
554  // variance, stdev
555  for (int ix=0;ix<cnt;++ix) {
556  for (int tid=0;tid<NUM_PROCESSES;++tid) {
557  int size = thread_data[tid].size[id];
558  if (ix >= size) continue;
559  auto data = thread_data[tid].get_ptr<T>(id);
560  if (data[ix] == 0) continue;
561  results[ix].variance += GSTATS_SQ(data[ix] - results[ix].avg);
562  }
563  if (results[ix].cnt > 1) results[ix].stdev = sqrt(results[ix].variance) / (results[ix].cnt - 1);
564  }
565  *num_indices = cnt;
566  return results;
567  }

618  template <typename T>
619  stat_metrics<long long> * get_histogram_log(const gstats_stat_id id, stat_metrics<T> * const metrics, const int num_metrics) {
620  // compute histogram
621  constexpr const int num_buckets = GSTATS_DEFAULT_HISTOGRAM_LOG_NUM_BUCKETS+1;
622  stat_metrics<long long> * __histogram = new stat_metrics<long long>[num_buckets+1];
623  acquireLock(&arrays_to_delete_lock);
624  arrays_to_delete.push_back((stat_metrics<double> *) __histogram);
625  releaseLock(&arrays_to_delete_lock);
626  for (int i=0;i<num_buckets+1;++i) {
627  memset(&__histogram[i], 0, sizeof(stat_metrics<long long>));
628  }
629  stat_metrics<long long> * histogram = &__histogram[1]; // handle -1s returned by log2_capped
630 
631  // parallel histogram construction
632  VERBOSE std::cout<<"parallel compute histogram: spawning "<<std::thread::hardware_concurrency()<<" threads"<<std::endl;
633  std::thread threads[std::thread::hardware_concurrency()];
634  volatile bool start = false;
635  for (int thread_id=0;thread_id<std::thread::hardware_concurrency();++thread_id) {
636  //cout<<" parallel compute histogram: spawning thread "<<thread_id<<std::endl;
637  threads[thread_id] = std::thread([thread_id, histogram, id, &start, num_buckets, metrics, this]() {
638 // while (!start) { __sync_synchronize(); }
639 
640  // compute our slice of the indices
641  int slice_size = this->num_indices[id] / std::thread::hardware_concurrency();
642  int start_ix = slice_size * thread_id;
643  int end_ix = (thread_id == std::thread::hardware_concurrency()-1) ? this->num_indices[id] : slice_size * (thread_id+1);
644 
645  stat_metrics<long long> * __thread_histogram = new stat_metrics<long long>[num_buckets+1];
646  for (int i=0;i<num_buckets+1;++i) {
647  memset(&__thread_histogram[i], 0, sizeof(stat_metrics<long long>));
648  }
649  stat_metrics<long long> * thread_histogram = &__thread_histogram[1]; // handle -1s returned by log2_capped
650 
651  // compute a thread-local histogram for only our slice of the indices
652  if (metrics) {
656  for (int ix=start_ix;ix<end_ix;++ix) {
657  ++thread_histogram[log2_capped(metrics[ix].first)].first;
658  ++thread_histogram[log2_capped(metrics[ix].cnt)].cnt;
659  ++thread_histogram[log2_capped(metrics[ix].min)].min;
660  ++thread_histogram[log2_capped(metrics[ix].max)].max;
661  ++thread_histogram[log2_capped(metrics[ix].sum)].sum;
662  ++thread_histogram[log2_capped(metrics[ix].avg)].avg;
663  ++thread_histogram[log2_capped(metrics[ix].variance)].variance;
664  ++thread_histogram[log2_capped(metrics[ix].stdev)].stdev;
665  }
666 
667  // use atomic primitives to add our histogram amounts to the shared array
668  for (int bucket=0;bucket<num_buckets;++bucket) {
669  __sync_fetch_and_add(&histogram[bucket].first, thread_histogram[bucket].first);
670  __sync_fetch_and_add(&histogram[bucket].cnt, thread_histogram[bucket].cnt);
671  __sync_fetch_and_add(&histogram[bucket].min, thread_histogram[bucket].min);
672  __sync_fetch_and_add(&histogram[bucket].max, thread_histogram[bucket].max);
673  __sync_fetch_and_add(&histogram[bucket].sum, thread_histogram[bucket].sum);
674  __sync_fetch_and_add(&histogram[bucket].avg, thread_histogram[bucket].avg);
675  __sync_fetch_and_add(&histogram[bucket].variance, thread_histogram[bucket].variance);
676  __sync_fetch_and_add(&histogram[bucket].stdev, thread_histogram[bucket].stdev);
677  }
678  } else {
682  for (int ix=start_ix;ix<end_ix;++ix) {
683  for (int tid=0;tid<NUM_PROCESSES;++tid) {
684  ++thread_histogram[log2_capped(get_stat<T>(tid, id, ix))].none;
685  }
686  }
687  // use atomic primitives to add our histogram amounts to the shared array
688  for (int bucket=0;bucket<num_buckets;++bucket) {
689  __sync_fetch_and_add(&histogram[bucket].none, thread_histogram[bucket].none);
690  }
691  }
692 
693  delete[] __thread_histogram;
694 
695  //cout<<" parallel compute histogram: thread "<<thread_id<<" terminated"<<std::endl;
696  });
697  //cout<<" parallel compute histogram: spawned thread "<<thread_id<<std::endl;
698  }
699  __sync_synchronize();
700  start = true;
701  __sync_synchronize();
702  VERBOSE std::cout<<"parallel compute histogram: joining "<<std::thread::hardware_concurrency()<<" threads"<<std::endl;
703  for (int i=0;i<std::thread::hardware_concurrency();++i) {
704  //cout<<" parallel compute histogram: joining thread "<<i<<std::endl;
705  threads[i].join();
706  //cout<<" parallel compute histogram: joined thread "<<i<<std::endl;
707  }
708  VERBOSE std::cout<<"parallel compute histogram: joined all."<<std::endl;
709 
710  return histogram;
711  }

836  }
837  }
838  }

967  }

969  case PRINT_HISTOGRAM_LOG:
