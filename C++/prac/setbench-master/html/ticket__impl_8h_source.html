59 static inline int
60 tl_trylock_version(volatile tl_t* tl, volatile tl_t* tl_old, int right) {
61  uint16_t version = tl_old->lr[right].version;
62  uint16_t one = (uint16_t) 1;
63  if (unlikely(version != tl_old->lr[right].ticket)) {
64  return 0;
65  }
66 
67 #if __GNUC__ >= 4 && __GNUC_MINOR__ >= 6
68  tl32_t tlo = {
69  {.version = version, .ticket = version}
70  }; //{ .version = version, .ticket = version};
71  tl32_t tln = {
72  {.version = version, .ticket = (uint16_t) (version + one)}
73  }; //{.version = version, .ticket = (version + 1)};
74  return CASV(&tl->lr[right].to_uint32, tlo.to_uint32, tln.to_uint32) == tlo.to_uint32;
75 #else
76  tl32_t tlo = {version, version};
77  tl32_t tln = {version, (uint16_t) (version + 1)};
78 #endif
79  return CASV(&tl->lr[right].to_uint32, tlo.to_uint32, tln.to_uint32) == tlo.to_uint32;
80 }

84 static inline int
85 tl_trylock_version_both(volatile tl_t* tl, volatile tl_t* tl_old) {
86  uint16_t v0 = tl_old->lr[0].version;
87  uint16_t v1 = tl_old->lr[1].version;
88  if (unlikely(v0 != tl_old->lr[0].ticket || v1 != tl_old->lr[1].ticket)) {
89  return 0;
90  }
91 
92 #if __GNUC__ >= 4 && __GNUC_MINOR__ >= 6
93  tl_t tlo = {.to_uint64 = tl_old->to_uint64};
94  return CASV(&tl->to_uint64, tlo.to_uint64, TLN_REMOVED) == tlo.to_uint64;
95 #else
96  /* tl_t tlo; */
97  /* tlo.uint64_t = tl_old->to_uint64; */
98  uint64_t tlo = *(uint64_t*) tl_old;
99 
100  return CASV((uint64_t*) tl, tlo, TLN_REMOVED) == tlo;
101 #endif

304 sval_t ticket<skey_t, sval_t, RecMgr>::bst_tk_delete(const int tid, skey_t key) {

307  node_t<skey_t, sval_t>* ppred = NULL;

309  uint64_t pred_ver = 0, ppred_ver = 0, right = 0, pright = 0;
310 
311 retry:
312 
313  { // reclamation guarded section
314  auto guard = recmgr->getGuard(tid);
315  curr = root;
316 
317  do {
318  curr_ver = curr->lock.to_uint64;
319 
320  ppred = pred;
321  ppred_ver = pred_ver;
322  pright = right;
323 
324  pred = curr;
325  pred_ver = curr_ver;
326 
327  if (key < curr->key) {
328  right = 0;
329  curr = curr->left;
330  } else {
331  right = 1;
332  curr = curr->right;
333  }
334  } while (likely(curr->left != NULL));
335 
336  if (curr->key != key) {
337  return NO_VALUE;
338  }
339 
340  if ((!tl_trylock_version(&ppred->lock, (volatile tl_t*) & ppred_ver, pright))) {
341  goto retry;
342  }
343 
344  if ((!tl_trylock_version_both(&pred->lock, (volatile tl_t*) & pred_ver))) {
345  tl_revert(&ppred->lock, pright);
346  goto retry;
347  }
348 
349  if (pright) {
350  if (right) {
351  ppred->right = pred->left;
352  } else {
353  ppred->right = pred->right;
354  }
355 
356  } else {
357  if (right) {
358  ppred->left = pred->left;
359  } else {
360  ppred->left = pred->right;
361  }
362  }
363 
364  tl_unlock(&ppred->lock, pright);
365 
366  recmgr->retire(tid, curr);
