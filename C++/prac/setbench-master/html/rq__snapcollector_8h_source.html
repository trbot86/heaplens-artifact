206  inline T linearize_update_at_write(
207  const int tid,

209  const T& lin_newval,
210  NodeType * const * const insertedNodes,
211  NodeType * const * const deletedNodes) {

224 #ifdef RQ_USE_TIMESTAMPS
225  if (pthread_rwlock_unlock(&rwlock)) setbench_error("could not read-unlock rwlock");
226 #endif
227 
228  if (insertedNodes[0]) insert_readonly_report_target_key(tid, insertedNodes[0]);
229  if (deletedNodes[0]) delete_report_target_key(tid, deletedNodes[0]);

241  const int tid,

245  NodeType * const * const insertedNodes,

262

265  if (deletedNodes[0]) delete_report_target_key(tid, deletedNodes[0]);

274  // invoke at the start of each traversal
275  inline void traversal_start(const int tid) {
276 #if !defined(RQ_USE_TIMESTAMPS)
277  threadData[tid].rq_lin_time = 1;
278 #endif
279 
280  threadData[tid].currentSnapCollector = snapPointer;
281  SOFTWARE_BARRIER;
282  if (!threadData[tid].currentSnapCollector->IsActive()) {
283  SnapCollector<NodeType,K> * candidate = recmgr->template allocate<SnapCollector<NodeType,K> >(tid);
284 #ifdef GSTATS_HANDLE_STATS
285  GSTATS_APPEND(tid, extra_type1_allocated_addresses, ((long long) candidate)%(1<<12));
286 #endif
287  candidate->init(tid, NUM_PROCESSES, recmgr, ds->KEY_MIN, ds->KEY_MAX+1);
288  if (__sync_bool_compare_and_swap(&snapPointer, threadData[tid].currentSnapCollector, candidate)) {
289  // delay retiring until later, because we've started accepting reports,
290  // and we don't want to waste time while we are accepting reports,
291  // because we don't want to receive many reports...
292  threadData[tid].snapCollectorToRetire = threadData[tid].currentSnapCollector;
293  threadData[tid].currentSnapCollector = candidate;
294  } else {
295  candidate->retire(tid, recmgr);
296  threadData[tid].currentSnapCollector = snapPointer;
297  }
298  }

307  // invoke at the end of each traversal:

311  void traversal_end(const int tid, K * const rqResultKeys, V * const rqResultValues, int * const startIndex, const K& lo, const K& hi) {
312  SnapCollector<NodeType,K> * sc = threadData[tid].currentSnapCollector;
313 
314  sc->BlockFurtherPointers(tid, recmgr);
315  SOFTWARE_BARRIER;
316  sc->Deactivate(NULL, NULL, NULL);
317  sc->BlockFurtherReports();
318  SOFTWARE_BARRIER;
319 
320  sc->Prepare(tid, recmgr);
321 
322  NodeType * curr = NULL;
323  while ((curr = sc->GetNext(tid))) {
324  if (curr->key < lo) continue;
325  if (curr->key > hi) break;
326  rqResultKeys[*startIndex] = curr->key;
327  rqResultValues[*startIndex] = curr->val;
328  ++*startIndex;
329  }
330 #if defined MICROBENCH
331  assert(*startIndex <= RQSIZE);
332 #endif
333 
334 #ifdef SNAPCOLLECTOR_PRINT_RQS
335 // for (int i=0;i<*startIndex;++i) {
336 // std::cout<<" "<<rqResultKeys[i];
337 // }
338 // std::cout<<std::endl;
339 #endif
340 
341  DEBUG_RECORD_RQ_SIZE(*startIndex);
342  DEBUG_RECORD_RQ_CHECKSUM(tid, threadData[tid].rq_lin_time, rqResultKeys, *startIndex);
343

345  if (threadData[tid].snapCollectorToRetire) {
346  threadData[tid].snapCollectorToRetire->retire(tid, recmgr);
