652  auto guard = recordmgr->getGuard(tid, true);
653  rqProvider->traversal_start(tid);
654 
655  // depth first traversal (of interesting subtrees)
656  int size = 0;
657  TRACE COUTATOMICTID("rangeQuery(lo="<<lo<<", hi="<<hi<<", size="<<(hi-lo+1)<<")"<<std::endl);
658 
659  stack.push(entry);
660  while (!stack.isEmpty()) {
661  Node<DEGREE,K> * node = stack.pop();
662  prefetch_range(node, sizeof(*node));
663  assert(node);
664 
665  // if leaf node, check if we should add its keys to the traversal
666  if (node->isLeaf()) {
667  rqProvider->traversal_try_add(tid, node, resultKeys, resultValues, &size, lo, hi);
668 
669  // else if internal node, explore its children
670  } else {
671  // find right-most sub-tree that could contain a key in [lo, hi]
672  int nkeys = node->getKeyCount();
673  int r = nkeys;
674  while (r > 0 && cmp(hi, (const K&) node->keys[r-1])) --r; // subtree rooted at node->ptrs[r] contains only keys > hi
675 
676  // find left-most sub-tree that could contain a key in [lo, hi]
677  int l = 0;
678  while (l < nkeys && !cmp(lo, (const K&) node->keys[l])) ++l; // subtree rooted at node->ptrs[l] contains only keys < lo
679 
680  // perform DFS from left to right (so push onto stack from right to left)
681  for (int i=r;i>=l; --i) stack.push(rqProvider->read_addr(tid, &node->ptrs[i]));

685 // stack.push(rqProvider->read_addr(tid, &node->ptrs[i]));
686 // }
687  }
688  }
689 
690  // success
691  rqProvider->traversal_end(tid, resultKeys, resultValues, &size, lo, hi);
692  return size;

872  n->leaf = false;
873  n->marked = false;
874  n->scxPtr = DUMMY;
875  n->searchKey = keys[size1];
876  n->size = 2;

880  // unless it is the root; this is because we test
881  // p == entry, above; in doing this, we are actually
882  // performing Root-Zero at the same time as this Overflow
883  // if n will become the root (of the B-slack tree)
884 
885  // construct info record to pass to SCX
886  info->numberOfNodes = 2;
887  info->numberOfNodesAllocated = 3;
888  info->numberOfNodesToFreeze = 1;
889  info->field = &p->ptrs[ixToL];
890  info->newNode = n;
891  info->insertedNodes[0] = n;
892  info->insertedNodes[1] = left;
893  info->insertedNodes[2] = right;
894  info->insertedNodes[3] = NULL;
895  info->deletedNodes[0] = l;
896  info->deletedNodes[1] = NULL;

898  if (scx(tid, info)) {

909  this->recordmgr->deallocate(tid, left);

911  }
912  }

1010 template <int DEGREE, typename K, class Compare, class RecManager>
1011 bool abtree_ns::abtree<DEGREE,K,Compare,RecManager>::fixWeightViolation(const int tid, Node<DEGREE,K>* viol) {
1012  if (viol->weight) return false;
1013 
1014  // assert: viol is internal (because leaves always have weight = 1)
1015  // assert: viol is not entry or root (because both always have weight = 1)
1016 
1017  // do an optimistic check to see if viol was already removed from the tree
1018  if (llx(tid, viol, NULL) == FINALIZED) {
1019  // recall that nodes are finalized precisely when
1020  // they are removed from the tree
1021  // we hand off responsibility for any violations at viol to the
1022  // process that removed it.
1023  return false;
1024  }
1025 
1026  wrapper_info<DEGREE,K> _info;
1027  wrapper_info<DEGREE,K>* info = &_info;
1028 
1029  // try to locate viol, and fix any weight violation at viol
1030  while (true) {
1031 // GSTATS_ADD_IX(tid, num_prop_guard_rebalance_attempts, 1, GSTATS_GET(tid, num_getguard));
1032 
1033  const K k = viol->searchKey;
1034  Node<DEGREE,K>* gp = NULL;
1035  Node<DEGREE,K>* p = entry;
1036  Node<DEGREE,K>* l = rqProvider->read_addr(tid, &p->ptrs[0]);
1037  prefetch_range(l, sizeof(*l));
1038  int ixToP = -1;
1039  int ixToL = 0;
1040  while (!l->isLeaf() && l != viol) {
1041  ixToP = ixToL;
1042  ixToL = l->getChildIndex(k, cmp);
1043  gp = p;
1044  p = l;
1045  l = rqProvider->read_addr(tid, &l->ptrs[ixToL]);
1046  prefetch_range(l, sizeof(*l));
1047  }
1048 
1049  if (l != viol) {
1050  // l was replaced by another update.
1051  // we hand over responsibility for viol to that update.
1052  return false;
1053  }
1054 
1055  // we cannot apply this update if p has a weight violation
1056  // so, we check if this is the case, and, if so, try to fix it
1057  if (!p->weight) {
1058  fixWeightViolation(tid, p);
1059  continue;
1060  }
1061 
1062  // perform LLXs
1063  if (!llx(tid, gp, NULL, 0, info->scxPtrs, info->nodes) || rqProvider->read_addr(tid, &gp->ptrs[ixToP]) != p) continue; // retry the search
1064  if (!llx(tid, p, NULL, 1, info->scxPtrs, info->nodes) || rqProvider->read_addr(tid, &p->ptrs[ixToL]) != l) continue; // retry the search
1065  if (!llx(tid, l, NULL, 2, info->scxPtrs, info->nodes)) continue; // retry the search
1066 
1067  const int c = p->getABDegree() + l->getABDegree();
1068  const int size = c-1;
1069 
1070  if (size <= b) {
1075  // create new node(s)
1076  // the new arrays are small enough to fit in a single node,
1077  // so we replace p by a new internal node.
1078  Node<DEGREE,K>* n = allocateNode(tid);
1079  arraycopy_ptrs(p->ptrs, 0, n->ptrs, 0, ixToL); // p and l are both internal, so we use arraycopy_ptrs
1080  arraycopy_ptrs(l->ptrs, 0, n->ptrs, ixToL, l->getABDegree());
1081  arraycopy_ptrs(p->ptrs, ixToL+1, n->ptrs, ixToL+l->getABDegree(), p->getABDegree()-(ixToL+1));
1082  arraycopy(p->keys, 0, n->keys, 0, ixToL);
1083  arraycopy(l->keys, 0, n->keys, ixToL, l->getKeyCount());
1084  arraycopy(p->keys, ixToL, n->keys, ixToL+l->getKeyCount(), p->getKeyCount()-ixToL);
1085  n->leaf = false; assert(!l->isLeaf());
1086  n->marked = false;
1087  n->scxPtr = DUMMY;
1088  n->searchKey = n->keys[0];
1089  n->size = size;
1090  n->weight = true;
1091 
1092  // construct info record to pass to SCX
1093  info->numberOfNodes = 3;
1094  info->numberOfNodesAllocated = 1;
1095  info->numberOfNodesToFreeze = 3;
1096  info->field = &gp->ptrs[ixToP];
1097  info->newNode = n;
1098 // info->insertedNodes[0] = info->deletedNodes[0] = NULL;
1099  info->insertedNodes[0] = n;
1100  info->insertedNodes[1] = NULL;
1101  info->deletedNodes[0] = p;
1102  info->deletedNodes[1] = l;
1103  info->deletedNodes[2] = NULL;
1104 
1105  if (scx(tid, info)) {
1106  TRACE COUTATOMICTID("absorb: SCX succeeded"<<std::endl);
1107 
1108  // absorb [check: slack@n]
1109  // no weight at pi(u)
1110  // degree at pi(u) -> eliminated
1111  // slack at pi(u) -> eliminated or slack at n
1112  // weight at u -> eliminated
1113  // no degree at u
1114  // slack at u -> slack at n
1115 
1120  fixDegreeViolation(tid, n);
1121  return true;
1122  }
1123  TRACE COUTATOMICTID("absorb: SCX FAILED"<<std::endl);
1124  this->recordmgr->deallocate(tid, n);
1125 
1126  } else {
1131  // merge keys of p and l into one big array (and similarly for children)
1132  // (we essentially replace the pointer to l with the contents of l)
1133  K keys[2*DEGREE];
1134  Node<DEGREE,K>* ptrs[2*DEGREE];
1135  arraycopy_ptrs(p->ptrs, 0, ptrs, 0, ixToL); // p and l are both internal, so we use arraycopy_ptrs
1136  arraycopy_ptrs(l->ptrs, 0, ptrs, ixToL, l->getABDegree());
1137  arraycopy_ptrs(p->ptrs, ixToL+1, ptrs, ixToL+l->getABDegree(), p->getABDegree()-(ixToL+1));
1138  arraycopy(p->keys, 0, keys, 0, ixToL);
1139  arraycopy(l->keys, 0, keys, ixToL, l->getKeyCount());
1140  arraycopy(p->keys, ixToL, keys, ixToL+l->getKeyCount(), p->getKeyCount()-ixToL);
1141 
1142  // the new arrays are too big to fit in a single node,
1143  // so we replace p by a new internal node and two new children.
1144  //
1145  // we take the big merged array and split it into two arrays,
1146  // which are used to create two new children u and v.
1147  // we then create a new internal node (whose weight will be zero
1148  // if it is not the root), with u and v as its children.
1149 
1150  // create new node(s)
1151  const int size1 = size / 2;
1152  Node<DEGREE,K>* left = allocateNode(tid);
1153  arraycopy(keys, 0, left->keys, 0, size1-1);
1154  arraycopy_ptrs(ptrs, 0, left->ptrs, 0, size1);
1155  left->leaf = false; assert(!l->isLeaf());
1156  left->marked = false;
1157  left->scxPtr = DUMMY;
1158  left->searchKey = keys[0];
1159  left->size = size1;
1160  left->weight = true;
1161 
1162  const int size2 = size - size1;
1163  Node<DEGREE,K>* right = allocateNode(tid);
1164  arraycopy(keys, size1, right->keys, 0, size2-1);
1165  arraycopy_ptrs(ptrs, size1, right->ptrs, 0, size2);
1166  right->leaf = false;
1167  right->marked = false;
1168  right->scxPtr = DUMMY;
1169  right->searchKey = keys[size1];
1170  right->size = size2;
1171  right->weight = true;
1172 
1173  Node<DEGREE,K>* n = allocateNode(tid);
1174  n->keys[0] = keys[size1-1];
1175  rqProvider->write_addr(tid, &n->ptrs[0], left);
1176  rqProvider->write_addr(tid, &n->ptrs[1], right);
1177  n->leaf = false;
1178  n->marked = false;
1179  n->scxPtr = DUMMY;
1180  n->searchKey = keys[size1-1]; // note: should be the same as n->keys[0]
1181  n->size = 2;
1182  n->weight = (gp == entry);
1183
