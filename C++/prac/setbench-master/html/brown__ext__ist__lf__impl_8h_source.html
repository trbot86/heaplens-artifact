847 template <typename K, typename V, class Interpolate, class RecManager>
848 void istree<K,V,Interpolate,RecManager>::helpFreeSubtree(const int tid, Node<K,V> * node) {
849  // if node is the root of a *large* subtree (256+ children),
850  // then have threads *collaborate* by reserving individual subtrees to free.
851  // idea: reserve a subtree before freeing it by CASing it to NULL
852  // we are done when all pointers are NULL.
853 
854  // conceptually you reserve the right to reclaim everything under a node
855  // (including the node) when you set its DIRTY_DIRTY_MARKED_FOR_FREE_MASK bit
856  //
857  // note: the dirty field doesn't exist for kvpair, value, empty value and rebuildop objects...
858  // so to reclaim those if they are children of the root node passed to this function,
859  // we claim the entire root node at the end, and go through those with one thread.
860 
861 #ifdef USE_GSTATS
862  TIMELINE_START(tid);
863  DURATION_START(tid);
864 #endif
865 
866  // first, claim subtrees rooted at CHILDREN of this node
867  // TODO: does this improve if we scatter threads in this iteration?
868  for (int i=0;i<node->degree;++i) {
869  auto ptr = prov->readPtr(tid, node->ptrAddr(i));
870  if (IS_NODE(ptr)) {
871  auto child = CASWORD_TO_NODE(ptr);
872  if (child == NULL) continue;
873 
874  // claim subtree rooted at child
875  while (true) {
876  auto old = child->dirty;
877  if (IS_DIRTY_MARKED_FOR_FREE(old)) break;
878  if (CASB(&child->dirty, old, old | DIRTY_MARKED_FOR_FREE_MASK)) {
879  freeSubtree(tid, ptr, true, false);
880  }
881  }
882  }
883  }
884 
885  // then try to claim the node itself to handle special object types (kvpair, value, empty value, rebuildop).
886  // claim node and its pointers that go to kvpair, value, empty value and rebuildop objects, specifically
887  // (since those objects, and their descendents in the case of a rebuildop object,
888  // are what remain unfreed [since all descendents of direct child *node*s have all been freed])
889  while (true) {
890  auto old = node->dirty;
891  if (IS_DIRTY_MARKED_FOR_FREE(old)) break;
892  if (CASB(&node->dirty, old, old | DIRTY_MARKED_FOR_FREE_MASK)) {
893  // clean up pointers to non-*node* objects (and descendents of such objects)
894  for (int i=0;i<node->degree;++i) {
895  auto ptr = prov->readPtr(tid, node->ptrAddr(i));
896  if (!IS_NODE(ptr)) {
897  freeSubtree(tid, ptr, true, false);
898  }
899  }
900  freeNode(tid, node, true); // retire the ACTUAL node
901  }
902  }
903 
904 #ifdef USE_GSTATS
905  DURATION_END(tid, duration_traverseAndRetire);
906  TIMELINE_END(tid, "freeSubtree");
907 #endif
908 }

911 size_t istree<K,V,Interpolate,RecManager>::markAndCount(const int tid, const casword_t ptr, bool tryTiming /* = true by default */) {
912 #ifdef MEASURE_DURATION_STATS
913  TimeThisScope obj (tid, duration_markAndCount, tryTiming);
914 #endif
915 
916  if (unlikely(IS_KVPAIR(ptr))) return 1;
917  if (unlikely(IS_VAL(ptr))) return 1 - IS_EMPTY_VAL(ptr);
918  if (unlikely(IS_REBUILDOP(ptr))) {
919  // if we are here seeing this rebuildop,
920  // then we ALREADY marked the node that points to the rebuildop,
921  // which means that rebuild op cannot possibly change that node
922  // to effect the rebuilding.
923  return markAndCount(tid, NODE_TO_CASWORD(CASWORD_TO_REBUILDOP(ptr)->rebuildRoot), false);
924  }
925 
926  assert(IS_NODE(ptr));
927  auto node = CASWORD_TO_NODE(ptr);
928 
929  // optimize by taking the sum from node->dirty if we run into a finished subtree
930  auto result = node->dirty;
931  if (IS_DIRTY_FINISHED(result)) return DIRTY_FINISHED_TO_SUM(result); // markAndCount has already FINISHED in this subtree, and sum is the count
932 
933  if (!IS_DIRTY_STARTED(result)) __sync_val_compare_and_swap(&node->dirty, 0, DIRTY_STARTED_MASK);
934 
935  // high level idea: if not at a leaf, try to divide work between any helpers at this node
936  // by using fetch&add to "soft-reserve" a subtree to work on.
937  // (each helper will get a different subtree!)
938  // note that all helpers must still try to help ALL subtrees after, though,
939  // since a helper might crash after soft-reserving a subtree.
940  // the DIRTY_FINISHED indicator makes these final helping attempts more efficient.
941  //
942  // this entire idea of dividing work between helpers first can be disabled
943  // by defining IST_DISABLE_COLLABORATIVE_MARK_AND_COUNT
944  //
945  // can the clean fetch&add work division be adapted better for concurrent ideal tree construction?
946  //
947  // note: could i save a second traversal to build KVPair arrays by having
948  // each thread call addKVPair for each key it sees in THIS traversal?
949  // (maybe avoiding sort order issues by saving per-thread lists and merging)
950 
951 #if !defined IST_DISABLE_COLLABORATIVE_MARK_AND_COUNT
952  // optimize for contention by first claiming a subtree to recurse on
953  // THEN after there are no more subtrees to claim, help (any that are still DIRTY_STARTED)
954  if (node->degree > MAX_ACCEPTABLE_LEAF_SIZE) { // prevent this optimization from being applied at the leaves, where the number of fetch&adds will be needlessly high
955  while (1) {
956  auto ix = __sync_fetch_and_add(&node->nextMarkAndCount, 1);
957  if (ix >= node->degree) break;
958  markAndCount(tid, prov->readPtr(tid, node->ptrAddr(ix)), false);
959 
960  auto result = node->dirty;
961  if (IS_DIRTY_FINISHED(result)) return DIRTY_FINISHED_TO_SUM(result); // markAndCount has already FINISHED in this subtree, and sum is the count
962  }
963  }
964 #endif
965 
966  // recurse over all subtrees
967  size_t keyCount = 0;
968  for (int i=0;i<node->degree;++i) {
969  keyCount += markAndCount(tid, prov->readPtr(tid, node->ptrAddr(i)), false);
970 
971  auto result = node->dirty;
972  if (IS_DIRTY_FINISHED(result)) return DIRTY_FINISHED_TO_SUM(result); // markAndCount has already FINISHED in this subtree, and sum is the count
973  }
974 
975  __sync_bool_compare_and_swap(&node->dirty, DIRTY_STARTED_MASK, SUM_TO_DIRTY_FINISHED(keyCount));
976  return keyCount;
977 }
