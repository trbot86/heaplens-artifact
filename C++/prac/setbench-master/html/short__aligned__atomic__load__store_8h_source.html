28 AO_INLINE unsigned short
29 AO_short_load(const volatile unsigned short *addr)
30 {
31  assert(((size_t)addr & (sizeof(unsigned short) - 1)) == 0);
32  /* Cast away the volatile for architectures like IA64 where */
33  /* volatile adds barrier semantics. */
34  return (*(unsigned short *)addr);
35 }
