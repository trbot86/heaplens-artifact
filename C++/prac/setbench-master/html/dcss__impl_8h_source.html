26 template <typename Unused>
27 dcssresult_t dcssProvider<Unused>::dcssHelp(const int tid, dcsstagptr_t tagptr, dcssptr_t snapshot, bool helpingOther) {
28  // figure out what the state should be
29  casword_t state = DCSS_STATE_FAILED;
30 
31  SOFTWARE_BARRIER;
32  casword_t val1 = *(snapshot->addr1);
33  SOFTWARE_BARRIER;
34 
35  //DELAY_UP_TO(1000);
36  if (val1 == snapshot->old1) { // linearize here(?)
37  state = DCSS_STATE_SUCCEEDED;
38  }
39 
40  // try to cas the state to the appropriate value
41  dcssptr_t ptr = TAGPTR_UNPACK_PTR(dcssDescriptors, tagptr);
42  casword_t retval;
43  bool failedBit;
44  MUTABLES_VAL_CAS_FIELD(failedBit, retval, ptr->mutables, snapshot->mutables, DCSS_STATE_UNDECIDED, state, DCSS_MUTABLES_MASK_STATE, DCSS_MUTABLES_OFFSET_STATE);
45  if (failedBit) return {DCSS_IGNORED_RETVAL,0}; // failed to access the descriptor: we must be helping another process complete its operation, so we will NOT use this return value!
46 
47  // TODO: do we do the announcement here? what will be announced exactly? do we let the user provide a pointer/value to announce as an argument to dcss? do we need to provide an operation to retrieve the current announcement for a given process?
48 
49  // finish the operation based on the descriptor's state
50  if ((retval == DCSS_STATE_UNDECIDED && state == DCSS_STATE_SUCCEEDED) // if we changed the state to succeeded OR
51  || retval == DCSS_STATE_SUCCEEDED) { // if someone else changed the state to succeeded
52 // if (state == DCSS_STATE_FAILED) DELAY_UP_TO(1000);
53  assert(helpingOther || ((snapshot->mutables & DCSS_MUTABLES_MASK_STATE) >> DCSS_MUTABLES_OFFSET_STATE) == DCSS_STATE_SUCCEEDED);
54  BOOL_CAS(snapshot->addr2, (casword_t) tagptr, snapshot->new2);
55  return {DCSS_SUCCESS,0};
56  } else { // either we or someone else changed the state to failed
57  assert((retval == DCSS_STATE_UNDECIDED && state == DCSS_STATE_FAILED)
58  || retval == DCSS_STATE_FAILED);
59  assert(helpingOther || ((snapshot->mutables & DCSS_MUTABLES_MASK_STATE) >> DCSS_MUTABLES_OFFSET_STATE) == DCSS_STATE_FAILED);
60  BOOL_CAS(snapshot->addr2, (casword_t) tagptr, snapshot->old2);
61 // if (state == DCSS_STATE_FAILED) DELAY_UP_TO(1000);
62  return {DCSS_FAILED_ADDR1,val1};
63  }
64 }

124 template <typename Unused>
125 dcssresult_t dcssProvider<Unused>::dcssPtr(const int tid, casword_t * addr1, casword_t old1, casword_t * addr2, casword_t old2, casword_t new2) {
126  // create dcss descriptor
127  dcssptr_t ptr = DESC_NEW(dcssDescriptors, DCSS_MUTABLES_NEW, tid);
128  assert((((dcssDescriptors[tid].mutables & MASK_SEQ) >> OFFSET_SEQ) & 1) == 0);
129  ptr->addr1 = addr1;
130  ptr->old1 = old1;
131  ptr->addr2 = addr2;
132  ptr->old2 = old2;
133  ptr->new2 = new2;
134 
135  DESC_INITIALIZED(dcssDescriptors, tid);
136 
137  // create tagptr
138  assert((((dcssDescriptors[tid].mutables & MASK_SEQ) >> OFFSET_SEQ) & 1) == 1);
139  tagptr_t tagptr = TAGPTR_NEW(tid, ptr->mutables, DCSS_TAGBIT);
140 
141  // perform the dcss operation described by our descriptor
142  casword_t r;
143  do {
144  assert(!isDcss(ptr->old2));
145  assert(isDcss(tagptr));
146  r = VAL_CAS(ptr->addr2, ptr->old2, (casword_t) tagptr);
147  if (isDcss(r)) {
148 #ifdef USE_DEBUGCOUNTERS
149  this->dcssHelpCounter->inc(tid);
150 #endif
151  dcssHelpOther(tid, (dcsstagptr_t) r);
152  }
153  } while (isDcss(r));
154  if (r == ptr->old2){
155 // DELAY_UP_TO(1000);
156  return dcssHelp(tid, tagptr, ptr, false); // finish our own operation
157  }
158  return {DCSS_FAILED_ADDR2,r};//DCSS_FAILED_ADDR2;
159 }
