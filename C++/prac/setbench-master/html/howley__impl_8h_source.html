238 template <typename skey_t, typename sval_t, class RecMgr>
239 find_result<sval_t> howley<skey_t, sval_t, RecMgr>::bst_find(const int tid, skey_t k, node_t<skey_t, sval_t> ** pred, operation_t<skey_t, sval_t> ** pred_op, node_t<skey_t, sval_t> ** curr, operation_t<skey_t, sval_t> ** curr_op, node_t<skey_t, sval_t> * aux_root, node_t<skey_t, sval_t> * root) {
240  find_result<sval_t> result;
241  skey_t curr_key;
242  node_t<skey_t, sval_t> * next;
243  node_t<skey_t, sval_t> * last_right;
244  operation_t<skey_t, sval_t> * last_right_op;
245 retry:
246  auto guard = recmgr->getGuard(tid, true);
247 
248  result.val = NO_VALUE;
249  result.code = NOT_FOUND_R;
250  *curr = aux_root;
251  *curr_op = (*curr)->op;
252  if (GETFLAG(*curr_op) != STATE_OP_NONE) {
253  if (aux_root == root) {
254  bst_help_child_cas(tid, UNFLAG(*curr_op), *curr);
255  goto retry;
256  } else {
257  result.code = ABORT;
258  return result;
259  }
260  }
261  next = (*curr)->right;
262  last_right = *curr;
263  last_right_op = *curr_op;
264  while (!ISNULL(next)) {
265  *pred = *curr;
266  *pred_op = *curr_op;
267  *curr = next;
268  *curr_op = (*curr)->op;
269 
270  if (GETFLAG(*curr_op) != STATE_OP_NONE) {
271  bst_help(tid, *pred, *pred_op, *curr, *curr_op);
272  goto retry;
273  }
274  curr_key = (*curr)->key;
275  if (k < curr_key) {
276  result.code = NOT_FOUND_L;
277  next = (*curr)->left;
278  } else if (k > curr_key) {
279  result.code = NOT_FOUND_R;
280  next = (*curr)->right;
281  last_right = *curr;
282  last_right_op = *curr_op;
283  } else {
284  result.val = (*curr)->value;
285  result.code = FOUND;
286  break;
287  }
288  }
289  if ((!(result.code == FOUND)) && (last_right_op != last_right->op)) goto retry;
290  if ((*curr)->op != *curr_op) goto retry;
291  return result;
292 }

295 sval_t howley<skey_t, sval_t, RecMgr>::bst_add(const int tid, skey_t k, sval_t v) {
296  node_t<skey_t, sval_t> * pred;
297  node_t<skey_t, sval_t> * curr;
298  operation_t<skey_t, sval_t> * pred_op;
299  operation_t<skey_t, sval_t> * curr_op;
300  while (true) {
301  auto guard = recmgr->getGuard(tid);
302 
303  auto result = bst_find(tid, k, &pred, &pred_op, &curr, &curr_op, root, root);
304  if (result.code == FOUND) return result.val;
305  bool is_left = (result.code == NOT_FOUND_L);
306  auto new_node = create_node(tid, k, v, NULL_NODEPTR, NULL_NODEPTR);
307  auto old = is_left ? curr->left : curr->right;
308  auto cas_op = alloc_op(tid);
309  cas_op->child_cas_op.is_left = is_left;
310  cas_op->child_cas_op.expected = old;
311  cas_op->child_cas_op.update = new_node;
312 
313  if (CASB(&curr->op, curr_op, FLAG(cas_op, STATE_OP_CHILDCAS))) {
314  bst_help_child_cas(tid, cas_op, curr);
315  return NO_VALUE;
316  } else {
317  recmgr->deallocate(tid, new_node);
318  recmgr->deallocate(tid, cas_op);
319  }
320  }
321 }

323 template <typename skey_t, typename sval_t, class RecMgr>
324 sval_t howley<skey_t, sval_t, RecMgr>::bst_remove(const int tid, skey_t k) {
325  node_t<skey_t, sval_t> * pred;
326  node_t<skey_t, sval_t> * curr;
327  node_t<skey_t, sval_t> * replace;
328  operation_t<skey_t, sval_t> * pred_op;
329  operation_t<skey_t, sval_t> * curr_op;
330  operation_t<skey_t, sval_t> * replace_op;
331  while (true) {
332  auto guard = recmgr->getGuard(tid);
333 
334  auto result = bst_find(tid, k, &pred, &pred_op, &curr, &curr_op, root, root);
335  if (!(result.code == FOUND)) return NO_VALUE;
336  if (ISNULL(curr->right) || ISNULL(curr->left)) {
337  if (CASB(&curr->op, curr_op, FLAG(curr_op, STATE_OP_MARK))) {
338  bst_help_marked(tid, pred, pred_op, curr);
339  recmgr->retire(tid, curr);
340  return result.val;
341  }
342  } else {
343  auto result2 = bst_find(tid, k, &pred, &pred_op, &replace, &replace_op, curr, root);
344  if ((result2.code == ABORT) || (curr->op != curr_op)) continue;
345  auto reloc_op = alloc_op(tid);
346  reloc_op->relocate_op.state = STATE_OP_ONGOING;
347  reloc_op->relocate_op.dest = curr;
348  reloc_op->relocate_op.dest_op = curr_op;
349  reloc_op->relocate_op.remove_key = k;
350  reloc_op->relocate_op.remove_value = result.val;
351  reloc_op->relocate_op.replace_key = replace->key;
352  reloc_op->relocate_op.replace_value = replace->value;
353 
354  if (CASB(&replace->op, replace_op, FLAG(reloc_op, STATE_OP_RELOCATE))) {
355  if (bst_help_relocate(tid, reloc_op, pred, pred_op, replace)) {
356  return result.val;
357  }
358  recmgr->retire(tid, reloc_op);
359  } else {
360  recmgr->deallocate(tid, reloc_op);
361  }
362  }
363  }
364 }

366 template <typename skey_t, typename sval_t, class RecMgr>
367 bool howley<skey_t, sval_t, RecMgr>::bst_help_relocate(const int tid, operation_t<skey_t, sval_t> * op, node_t<skey_t, sval_t> * pred, operation_t<skey_t, sval_t> * pred_op, node_t<skey_t, sval_t> * curr) {
368  int seen_state = op->relocate_op.state;
369  if (seen_state == STATE_OP_ONGOING) {
370  auto seen_op = CASV(&op->relocate_op.dest->op, op->relocate_op.dest_op, FLAG(op, STATE_OP_RELOCATE));
371  if ((seen_op == op->relocate_op.dest_op) || (seen_op == FLAG(op, STATE_OP_RELOCATE))) {
372  CASV(&op->relocate_op.state, STATE_OP_ONGOING, STATE_OP_SUCCESSFUL);
373  seen_state = STATE_OP_SUCCESSFUL;
374  } else {
375  seen_state = CASV(&op->relocate_op.state, STATE_OP_ONGOING, STATE_OP_FAILED);
376  }
377  }
378  if (seen_state == STATE_OP_SUCCESSFUL) {
379  CASB(&op->relocate_op.dest->key, op->relocate_op.remove_key, op->relocate_op.replace_key);
380  CASB(&op->relocate_op.dest->value, op->relocate_op.remove_value, op->relocate_op.replace_value);
381  CASB(&op->relocate_op.dest->op, FLAG(op, STATE_OP_RELOCATE), FLAG(op, STATE_OP_NONE));
382  }
383  bool result = (seen_state == STATE_OP_SUCCESSFUL);
384  if (op->relocate_op.dest == curr) return result;
385  CASB(&curr->op, FLAG(op, STATE_OP_RELOCATE), FLAG(op, result ? STATE_OP_MARK : STATE_OP_NONE));
386  if (result) {
387  if (op->relocate_op.dest == pred) pred_op = FLAG(op, STATE_OP_NONE);
388  bst_help_marked(tid, pred, pred_op, curr);
389  }
390  return result;
391 }
