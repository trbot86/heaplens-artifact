47 // parse token starting at argv[ix],
48 // place bindings for the token at the end of customBinding, and
49 // return the index of the first character in the next token,
50 // or the size of the std::string argv if there are no further tokens.
51 static unsigned parseToken(std::string argv, int ix) {
52  // token is of one of following forms:
53  // INT
54  // INT-INT
55  // and is either followed by "," is the end of the string.
56  // we first determine which is the case
57 
58  // read first INT
59  int ix2 = ix;
60  while (ix2 < argv.size() && argv[ix2] != '.') ++ix2;
61  std::string token = argv.substr(ix, ix2-ix+1);
62  int a = atoi(token.c_str());
63 
64  // check if the token is of the first form: INT
65  ix = ix+digits(a); // first character AFTER first INT
66  if (ix >= argv.size() || argv[ix] == '.') {
67 
68  // add single binding
69  //cout<<"a="<<a<<std::endl;
70  customBinding[numCustomBindings++] = a;
71  assert(numCustomBindings <= MAX_THREADS_POW2);
72 
73  // token is of the second form: INT-INT
74  } else {
75  assert(argv[ix] == '-');
76  ++ix; // skip '-'
77 
78  // read second INT
79  token = argv.substr(ix, ix2-ix+1);
80  int b = atoi(token.c_str());
81  //cout<<"a="<<a<<" b="<<b<<std::endl;
82 
83  // add range of bindings
84  for (int i=a;i<=b;++i) {
85  customBinding[numCustomBindings++] = i;
86  assert(numCustomBindings <= MAX_THREADS_POW2);
87  }
88 
89  ix = ix+digits(b); // first character AFTER second INT
90  }
91  // note: ix is the first character AFTER the last INT in the token
92  // this is either a comma ('.') or the end of the std::string argv.
93  return (ix >= argv.size() ? argv.size() : ix+1 /* skip '.' */);
94 }
