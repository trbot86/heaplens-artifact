221  // for all schemes except reference counting
222  inline void retire(const int tid, T* p) {
223  threadData[tid].currentBag->add(p);
224  DEBUG2 this->debug->addRetired(tid, 1);
225  if (threadData[tid].currentBag->getSizeInBlocks() >= 2) {
226  // only execute the following logic once every X times (starting at 0) we see that our current bag is too large
227  // (resetting the count when we rotate bags).
228  // if we are being prevented from reclaiming often, then we will quickly (within X operations) scan all threads.
229  // otherwise, we will avoid scanning all threads too often.
230  if ((++threadData[tid].timesBagTooLargeSinceRotation) % 1000) return;
231 
232  long readEpoch = epoch;
233  const long ann = threadData[tid].localvar_announcedEpoch;
234 
235  // if our announced epoch is different from the current epoch, skip the following, since we're going to rotate limbo bags on our next operation, anyway
236  if (readEpoch != BITS_EPOCH(ann)) return;
237 
238  // scan all threads (skipping any threads we've already checked) to see if we can advance the epoch (and subsequently reclaim memory)
239  for (; threadData[tid].checked < this->NUM_PROCESSES; ++threadData[tid].checked) {
240  const int otherTid = threadData[tid].checked;
241  long otherAnnounce = threadData[otherTid].announcedEpoch.load(std::memory_order_relaxed);
242  if (!(BITS_EPOCH(otherAnnounce) == readEpoch || QUIESCENT(otherAnnounce))) return;
243  }
244  __sync_bool_compare_and_swap(&epoch, readEpoch, readEpoch+EPOCH_INCREMENT);
245  }
