12  * Invoke the macro GSTATS_DECLARE_STATS_OBJECT to declare a gstats_t object
13  * (and make it globally visible).
14  * Invoke GSTATS_DECLARE_ALL_STAT_IDS to declare (and make globally visible)
15  * integer identifiers for each statistic defined in GSTATS_HANDLE_STATS.
16  * Invoke GSTATS_CREATE_ALL at run time before starting your experiments.
17  * Invoke GSTATS_ADD and/or GSTATS_ADD_IX to add some value to a statistic
18  * (when using statistics with multiple indices, for example, for different
19  * times in an execution, use GSTATS_ADD_IX to add to a specific index)
20  * You can clear all gathered stats by invoking GSTATS_CLEAR_ALL.
21  * Print all stats by invoking GSTATS_PRINT.
22  *
23  * Example GSTATS_HANDLE_STATS definition:
24  * #define GSTATS_HANDLE_STATS(gstats_handle_stat) \
25  * gstats_handle_stat(visited_in_bags, 1, {gstats_output_item(PRINT_HISTOGRAM_LOG, SUM, BY_INDEX)}) \
26  * gstats_handle_stat(insertions, 1, {gstats_output_item(PRINT_RAW, SUM, TOTAL)}) \
27  * gstats_handle_stat(insertions_stdev, 1, {gstats_output_item(PRINT_RAW, STDEV, TOTAL)}) \
28  * gstats_handle_stat(insertions_per_thread, num_threads, {gstats_output_item(PRINT_RAW, SUM, BY_THREAD)}) \
29  * gstats_handle_stat(tree_size_over_time_histogram, 1000000, {gstats_output_item(PRINT_HISTOGRAM_LIN, SUM, BY_INDEX, 10)})
