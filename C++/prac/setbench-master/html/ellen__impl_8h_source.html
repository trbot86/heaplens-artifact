207 template <typename skey_t, typename sval_t, class RecMgr>
208 sval_t ellen<skey_t, sval_t, RecMgr>::bst_insert(const int tid, const skey_t key, const sval_t value) {
209  while (1) {
210  auto guard = recmgr->getGuard(tid);
211 
212  auto p = root;
213  auto pupdate = p->update;
214  SOFTWARE_BARRIER;
215  auto l = p->left;
216  while (l->left) {
217  p = l;
218  pupdate = p->update;
219  SOFTWARE_BARRIER;
220  l = (key < l->key) ? l->left : l->right;
221  }
222  if (l->key == key) {
223  return l->value;
224  }
225  if (GETFLAG(pupdate) != STATE_CLEAN) {
226  bst_help(tid, pupdate);
227  } else {
228  auto new_node = create_node(tid, key, value, NULL, NULL);
229  auto new_sibling = create_node(tid, l->key, l->value, NULL, NULL);
230  auto new_internal = (key < l->key)
231  ? create_node(tid, l->key, NO_VALUE, new_node, new_sibling)
232  : create_node(tid, key, NO_VALUE, new_sibling, new_node);
233  auto op = create_iinfo_t(tid, p, new_internal, l);
234  auto result = CASV(&p->update, pupdate, FLAG(op, STATE_IFLAG));
235  if (result == pupdate) {
236  bst_help_insert(tid, op);
237  return NO_VALUE;
238  } else {
239  recmgr->deallocate(tid, new_node);
240  recmgr->deallocate(tid, new_sibling);
241  recmgr->deallocate(tid, new_internal);
242  recmgr->deallocate(tid, op);
243  bst_help(tid, result);
244  }
245  }
246  }

250 sval_t ellen<skey_t, sval_t, RecMgr>::bst_delete(const int tid, skey_t key) {
251  while (1) {

253 
254  node_t<skey_t, sval_t> * gp = NULL;
255  info_t<skey_t, sval_t> * gpupdate = NULL;
256  auto p = root;
257  auto pupdate = p->update;
258  SOFTWARE_BARRIER;
259  auto l = p->left;
260  while (l->left) {
261  gp = p;
262  p = l;
263  gpupdate = pupdate;
264  pupdate = p->update;
265  SOFTWARE_BARRIER;
266  l = (key < l->key) ? l->left : l->right;
267  }
268  if (l->key != key) {
269  return NO_VALUE;
270  }
271  auto found_value = l->value;
272  if (GETFLAG(gpupdate) != STATE_CLEAN) {
273  bst_help(tid, gpupdate);
274  } else if (GETFLAG(pupdate) != STATE_CLEAN) {
275  bst_help(tid, pupdate);
276  } else {
277  auto op = create_dinfo_t(tid, gp, p, l, pupdate);
278  auto result = CASV(&gp->update, gpupdate, FLAG(op, STATE_DFLAG));
279  if (result == gpupdate) {
280  if (bst_help_delete(tid, op)) return found_value;
281  } else {
282  recmgr->deallocate(tid, op);
283  bst_help(tid, result);
284  }
285  }
