1 /*
2  * Modular implementation of LLX and SCX.
3  *
4  * These primitives were introduced in my PODC'13 paper:
5  * Pragmatic primitives for non-blocking data structures.
6  *
7  * This implementation applies the weak descriptor optimization described in my DISC 2017 paper:
8  * Reuse, don't recycle -- transforming lock-free algorithms that throw away descriptors.
9  *
10  * Usage:
11  * 1. Include this header.
12  * 2. Create an SCXProvider object, specifying the template parameter
13  * MaxNodeDependenciesPerSCX, which should be an upper bound on the
14  * maximum number of nodes that any single SCX operation will depend on.
15  * 3. In an update that will perform SCX, you will ultimately invoke scxExecute
16  * to perform an SCX. Before you can do this, you must invoke scxInit,
17  * and then perform one or more invocations of scxAddNode to add nodes
18  * that the SCX operation will depend on. Before performing
19  * scxAddNode(tid, node, ...), you must perform LLX(tid, node, ...),
20  * and the return value of this LLX must satisfy isSuccessfulLLXResult.
21  * The return value of this LLX is then fed to scxAddNode.
22  * After performing your successful LLX(s), scxInit and scxAddNode(s),
23  * you can invoke scxExecute, which will return true if it succeeded,
24  * and false otherwise.
25  *
26  * Author: Trevor Brown (C) 2018
27  * Created on April 4, 2018, 4:11 PM
28  */

94  int help(const int tid, tagptr_t const tagptr, SCXRecord const * const snap, bool helpingOther) {
95  SCXRecord *ptr = TAGPTR1_UNPACK_PTR(tagptr);
96  for (int i=helpingOther; i<snap->c.numNodes; ++i) {
97  bool successfulCAS = __sync_bool_compare_and_swap(&snap->c.nodes[i]->scxPtr, (void *) snap->c.scxPtrsSeen[i], tagptr);
98  auto exp = snap->c.nodes[i]->scxPtr;
99  if (successfulCAS || exp == (scx_handle_t) tagptr) continue; // if node is already frozen for our operation
100 
101  // note: we can get here only if:
102  // 1. the state is inprogress, and we just failed a cas, and every helper will fail that cas (or an earlier one), so the scx must abort, or
103  // 2. the state is committed or aborted
104  // (this suggests that it might be possible to get rid of the allFrozen bit)
105 
106  // read mutable allFrozen field of descriptor
107  bool succ;
108  bool allFrozen = DESC1_READ_FIELD(succ, ptr->c.mutables, tagptr, MUTABLES1_MASK_ALLFROZEN, MUTABLES1_OFFSET_ALLFROZEN);
109  if (!succ) return SCXRecord::STATE_ABORTED;
110 
111  if (allFrozen) {
112  return SCXRecord::STATE_COMMITTED;
113  } else {
114  const int newState = SCXRecord::STATE_ABORTED;
115  MUTABLES1_WRITE_FIELD(ptr->c.mutables, snap->c.mutables, newState, MUTABLES1_MASK_STATE, MUTABLES1_OFFSET_STATE);
116  return newState;
117  }
118  }
119  MUTABLES1_WRITE_BIT(ptr->c.mutables, snap->c.mutables, MUTABLES1_MASK_ALLFROZEN);
120  SOFTWARE_BARRIER;
121  for (int i=0; i<snap->c.numNodes; ++i) {
122  if (!snap->c.finalize[i]) continue;
123  snap->c.nodes[i]->marked = true;
124  }
125 
126  // CAS in the new sub-tree (update CAS)
127  __sync_bool_compare_and_swap(snap->c.field, snap->c.oldVal, snap->c.newVal);
128 
129  MUTABLES1_WRITE_FIELD(ptr->c.mutables, snap->c.mutables, SCXRecord::STATE_COMMITTED, MUTABLES1_MASK_STATE, MUTABLES1_OFFSET_STATE);
130  return SCXRecord::STATE_COMMITTED; // success
131  }

133  void helpOther(const int tid, tagptr_t tagptr) {
134  if (tagptr == INIT_SCX_HANDLE) {
135  return; // deal with the dummy descriptor
136  }
137  SCXRecord snap;
138  if (DESC1_SNAPSHOT(&snap, tagptr, SCXRecord::size)) {
139  assert((UNPACK1_SEQ(tagptr) & 0x1));
140  assert((UNPACK1_SEQ(snap.c.mutables) & 0x1));
141  assert(UNPACK1_SEQ(tagptr) == UNPACK1_SEQ(snap.c.mutables));
142  help(tid, tagptr, &snap, true);
143  }
144  }

167  scx_handle_t llx(const int tid, NodeT const * const srcNode, NodeT * const dest = NULL, const int bytesToCopy = sizeof(NodeT)) {
168  const bool marked = srcNode->marked;
169  SOFTWARE_BARRIER;
170  tagptr_t tagptr = (tagptr_t) srcNode->scxPtr;
171 
172  // read mutable state field of descriptor
173  bool succ;
174  int state = DESC1_READ_FIELD(succ, TAGPTR1_UNPACK_PTR(tagptr)->c.mutables, tagptr, MUTABLES1_MASK_STATE, MUTABLES1_OFFSET_STATE);
175  if (!succ) state = SCXRecord::STATE_COMMITTED;
176  // note: special treatment for alg in the case where the descriptor has already been reallocated (impossible before the transformation, assuming safe memory reclamation)
177  SOFTWARE_BARRIER;
178 
179  if (state == SCXRecord::STATE_ABORTED || ((state == SCXRecord::STATE_COMMITTED) && !srcNode->marked)) {
180  // read snapshot fields (if outputSnapshot is non-NULL)
181  if (dest) memcpy(dest, srcNode, bytesToCopy);
182  if ((tagptr_t) srcNode->scxPtr == tagptr) return (scx_handle_t) tagptr; // we have a snapshot
183  }
184 
185  if (state == SCXRecord::STATE_INPROGRESS) {
186  helpOther(tid, tagptr);
187  }
188  return (marked ? FINALIZED : FAILED);
189  }

201  inline void scxAddNode(const int tid, NodeT * const node, bool finalize, scx_handle_t const llxResult) {
202  SCXRecord * scxptr = &DESC1_ARRAY[tid];
203  assert(!(UNPACK1_SEQ(scxptr->c.mutables) & 0x1));
204  int * ix = &scxptr->c.numNodes;
205  scxptr->c.nodes[*ix] = node;
206  scxptr->c.finalize[*ix] = finalize;
207  scxptr->c.scxPtrsSeen[*ix] = llxResult;
208  ++(*ix);
209  }

211  inline bool scxExecute(const int tid, void * volatile * const field, void * const oldVal, void * const newVal) {
212  SCXRecord * scxptr = &DESC1_ARRAY[tid];
213  scxptr->c.oldVal = oldVal;
214  scxptr->c.newVal = newVal;
215  scxptr->c.field = field;
216  DESC1_INITIALIZED(tid);
217  auto tagptr = TAGPTR1_NEW(tid, scxptr->c.mutables);
218  assert((UNPACK1_SEQ(tagptr) & 0x1));
219  auto result = help(tid, tagptr, scxptr, false);
220  DESC1_NEW(tid);
221  assert(!(UNPACK1_SEQ(scxptr->c.mutables) & 0x1));
222  return (result == SCXRecord::STATE_COMMITTED);
223  }
