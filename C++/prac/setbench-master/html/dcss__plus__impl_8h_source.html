25 template <typename PAYLOAD_T>
26 dcsspresult_t dcsspProvider<PAYLOAD_T>::dcsspHelp(const int tid, dcssptagptr_t tagptr, dcsspptr_t snapshot, bool helpingOther) {
27  // figure out what the state should be
28  casword_t state = DCSSP_STATE_FAILED;
29 
30  SOFTWARE_BARRIER;
31  casword_t val1 = *(snapshot->addr1);
32  SOFTWARE_BARRIER;
33 
34  //DELAY_UP_TO(1000);
35  if (val1 == snapshot->old1) { // linearize here(?)
36  state = DCSSP_STATE_SUCCEEDED;
37  }
38 
39  // try to cas the state to the appropriate value
40  dcsspptr_t ptr = TAGPTR_UNPACK_PTR(dcsspDescriptors,tagptr);
41  casword_t retval;
42  bool failedBit;
43  MUTABLES_VAL_CAS_FIELD(failedBit, retval, ptr->mutables, snapshot->mutables, DCSSP_STATE_UNDECIDED, state, DCSSP_MUTABLES_MASK_STATE, DCSSP_MUTABLES_OFFSET_STATE);
44  if (failedBit) return {DCSSP_IGNORED_RETVAL,0}; // failed to access the descriptor: we must be helping another process complete its operation, so we will NOT use this return value!
45 
46  // TODO: do we do the announcement here? what will be announced exactly? do we let the user provide a pointer/value to announce as an argument to dcssp? do we need to provide an operation to retrieve the current announcement for a given process?
47 
48  // finish the operation based on the descriptor's state
49  if ((retval == DCSSP_STATE_UNDECIDED && state == DCSSP_STATE_SUCCEEDED) // if we changed the state to succeeded OR
50  || retval == DCSSP_STATE_SUCCEEDED) { // if someone else changed the state to succeeded
51 // if (state == DCSSP_STATE_FAILED) DELAY_UP_TO(1000);
52  assert(helpingOther || ((snapshot->mutables & DCSSP_MUTABLES_MASK_STATE) >> DCSSP_MUTABLES_OFFSET_STATE) == DCSSP_STATE_SUCCEEDED);
53  BOOL_CAS(snapshot->addr2, (casword_t) tagptr, snapshot->new2);
54  return {DCSSP_SUCCESS,0};
55  } else { // either we or someone else changed the state to failed
56  assert((retval == DCSSP_STATE_UNDECIDED && state == DCSSP_STATE_FAILED)
57  || retval == DCSSP_STATE_FAILED);
58  assert(helpingOther || ((snapshot->mutables & DCSSP_MUTABLES_MASK_STATE) >> DCSSP_MUTABLES_OFFSET_STATE) == DCSSP_STATE_FAILED);
59  BOOL_CAS(snapshot->addr2, (casword_t) tagptr, snapshot->old2);
60 // if (state == DCSSP_STATE_FAILED) DELAY_UP_TO(1000);
61  return {DCSSP_FAILED_ADDR1,val1};
62  }
63 }

128 template <typename PAYLOAD_T>
129 dcsspresult_t dcsspProvider<PAYLOAD_T>::dcsspPtr(const int tid, casword_t * addr1, casword_t old1, casword_t * addr2, casword_t old2, casword_t new2, PAYLOAD_T * const payload1, PAYLOAD_T * const payload2) {
130  // create dcssp descriptor
131  dcsspptr_t ptr = DESC_NEW(dcsspDescriptors, DCSSP_MUTABLES_NEW, tid);
132  assert((((dcsspDescriptors[tid].mutables & MASK_SEQ) >> OFFSET_SEQ) & 1) == 0);
133  ptr->addr1 = addr1;
134  ptr->old1 = old1;
135  ptr->addr2 = addr2;
136  ptr->old2 = old2;
137  ptr->new2 = new2;
138 
139  // add payload1 and payload2 to the dcssp descriptor
140  int i;
141  for (i=0;payload1[i];++i) {
142  ptr->payload1[i] = payload1[i];
143  assert(i < MAX_PAYLOAD_PTRS);
144  }
145  ptr->payload1[i] = NULL;
146  for (i=0;payload2[i];++i) {
147  ptr->payload2[i] = payload2[i];
148  assert(i < MAX_PAYLOAD_PTRS);
149  }
150  ptr->payload2[i] = NULL;
151  DESC_INITIALIZED(dcsspDescriptors, tid);
152 
153  // create tagptr
154  assert((((dcsspDescriptors[tid].mutables & MASK_SEQ) >> OFFSET_SEQ) & 1) == 1);
155  tagptr_t tagptr = TAGPTR_NEW(tid, ptr->mutables, DCSSP_TAGBIT);
156 
157  // perform the dcssp operation described by our descriptor
158  casword_t r;
159  do {
160  assert(!isDcssp(ptr->old2));
161  assert(isDcssp(tagptr));
162  r = VAL_CAS(ptr->addr2, ptr->old2, (casword_t) tagptr);
163  if (isDcssp(r)) {
164 #ifdef USE_DEBUGCOUNTERS
165  this->dcsspHelpCounter->inc(tid);
166 #endif
167  dcsspHelpOther(tid, (dcssptagptr_t) r);
168  }
169  } while (isDcssp(r));
170  if (r == ptr->old2){
171 // DELAY_UP_TO(1000);
172  return dcsspHelp(tid, tagptr, ptr, false); // finish our own operation
173  }
174  return {DCSSP_FAILED_ADDR2,r};//DCSSP_FAILED_ADDR2;
175 }
