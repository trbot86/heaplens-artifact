152  // retire report items
153  for (int i=0;i<NUM_THREADS;++i) {
154  ReportItem * curr = reportHeads[i*PREFETCH_SIZE_WORDS];
155  while (curr != NULL) {
156  if (curr != blocker) recmgr->retire(tid, curr); // blocker can exist in many per-thread lists, but we only want to retire it once, below.

159  }

161  recmgr->retire(tid, blocker);

165  NodeWrapper * curr = this->tail;
166  if (isBlocker(curr)) {

338  NodeType * GetNext(int tid) {
339  NodeWrapper * currLoc = (NodeWrapper *) currLocations[tid*PREFETCH_SIZE_WORDS];
340  int currRepLoc = currRepLocations[tid*PREFETCH_SIZE_WORDS];
341  std::vector<CompactReportItem *> * allReports = gAllReports;
342 
343  while (true) {
344  CompactReportItem * rep = NULL;
345  K repKey = KEY_MAX;
346  if (allReports->size() > currRepLoc) {
347  rep = (*allReports)[currRepLoc];
348  repKey = rep->key;
349  }
350  K nodeKey = KEY_MAX;
351  NodeWrapper * next = currLoc->next;
352  if (next != NULL) {
353  nodeKey = next->key;
354  }
355 
356  // Option 1: node key < rep key. Return node.
357  if (nodeKey < repKey) {
358  currLocations[tid*PREFETCH_SIZE_WORDS] = next;
359  currRepLocations[tid*PREFETCH_SIZE_WORDS] = currRepLoc;
360  return next->node;
361  }
362 
363  // Option 2: node key == rep key
364  if (nodeKey == repKey) {
365  // 2.a - both are infinity - iteration done.
366  if (nodeKey == KEY_MAX) {
367  currLocations[tid*PREFETCH_SIZE_WORDS] = currLoc;
368  currRepLocations[tid*PREFETCH_SIZE_WORDS] = currRepLoc;
369  return NULL;
370  }
371  // node and report with the same key ::
372 
373  // skip not-needed reports
374  while (currRepLoc + 1 < allReports->size()) {
375  CompactReportItem * nextRep = (*allReports)[currRepLoc + 1];
376  // dismiss a duplicate, or an insert followed by a matching delete:
377  if (rep->key == nextRep->key && rep->node == nextRep->node) {
378  currRepLoc++;
379  rep = nextRep;
380  } else {
381  break;
382  }
383  }
384  // standing on an insert report to a node I am holding:
385  // 1. Return the current node.
386  // 2. Skip over rest of reports for that key.
387  if (rep->t == ReportType::Add && (NodeType *) rep->node == next->node) {
388  while (currRepLoc < allReports->size()
389  && (*allReports)[currRepLoc]->key == rep->key) {
390  currRepLoc++;
391  }
392  currRepLocations[tid*PREFETCH_SIZE_WORDS] = currRepLoc;
393  currLocations[tid*PREFETCH_SIZE_WORDS] = next;
394  return next->node;
395  }
396  // standing on an insert report to a different node than I hold:
397  // 1. Return the reported node.
398  // 2. Skip over rest of reports for that key.
399  if (rep->t == ReportType::Add && (NodeType *) rep->node != next->node) {
400  NodeType * returnValue = (NodeType *) rep->node;
401  while (currRepLoc < allReports->size()
402  && (*allReports)[currRepLoc]->key == rep->key) {
403  currRepLoc++;
404  }
405  currRepLocations[tid*PREFETCH_SIZE_WORDS] = currRepLoc;
406  currLocations[tid*PREFETCH_SIZE_WORDS] = next;
407  return returnValue;
408  }
409  // standing on a delete report to a different node than I hold:
410  // skip over it and continue the big loop.
411  if (rep->t == ReportType::Remove && (NodeType *) rep->node != next->node) {
412  currRepLoc++;
413  continue;
414  }
415  // standing on a delete report to the node that I hold:
416  // 1. advance over the node that I hold.
417  // 2. advance with the report.
418  // 3. continue the bigloop
419  currLoc = next;
420  currRepLoc++;
421  continue;
422  }
423 
424  // Option 3: node key > rep key
425  if (nodeKey > repKey) {
426  // skip not-needed reports
427  while (currRepLoc + 1 < allReports->size()) {
428  CompactReportItem * nextRep = (*allReports)[currRepLoc + 1];
429  // dismiss a duplicate, or an insert followed by a matching delete:

431  currRepLoc++;

434  break;
435  }
436  }

439  currRepLoc++;

444  // 1. skip over rest of the reports for the same key.
