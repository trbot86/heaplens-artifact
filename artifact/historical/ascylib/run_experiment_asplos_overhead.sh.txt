#!/bin/bash

UPDATE=0
expLabel=""

while [ $# -gt 0 ]; do
    case $1 in
        -u | --update)
            if [[ -z "$2" || "$2" == -* ]]; then
                echo "Must specify an update percentage with option -u/--update." >&2
                exit 1
            fi
            UPDATE=$2
            shift
        ;;
        -l | --label)
            if [[ -z "$2" || "$2" == -* ]]; then
                echo "Must specify a label with option -l/--label." >&2
                exit 1
            fi
            expLabel=$2
            shift
        ;;
    esac
    shift
done

echo "Setting LD_LIBRARY_PATH..."
LD_LIBRARY_PATH=./external/lib/
export LD_LIBRARY_PATH

dirLabels=("throughput" "cache-miss-rate" "page-faults" "L1-dcache-load-miss-rate" "LLC-load-miss-rate" "LLC-store-miss-rate" "context-switch-rate" "dTLB-load-miss-rate")

for label in "${dirLabels[@]}"
do
    if [ ! -d exp_"$label"_u"$UPDATE"_"$expLabel" ]; then
        echo "Making directory for $label"
        mkdir exp_"$label"_u"$UPDATE"_overhead_"$expLabel"
    fi
    # rm exp_"$label"_u"$UPDATE"_"$expLabel"/*
done

dataStructures=("ellen" "drachsler" "bronson" "ellen-overhead" "drachsler-overhead" "bronson-overhead")

THRU_REGEX="#Mops ([\.0-9]+).*"
PERF_REGEX="\s*([0-9]+)\s+([-a-zA-Z0-9]+).*"

processLine() {
    line="$1"
    line="${line//,/}"
    echo $line
    if [[ $line =~ $THRU_REGEX ]]
    then
        # touch exp_throughput/u"$UPDATE".txt
        NUM_OPS=$(echo "${BASH_REMATCH[1]}" | awk '{printf "%d", $1 * 1000000}')
        echo "$DATA_STRUCTURE $NUM_THREADS $NUM_OPS" >> /root/sifter/ASCYLIB/exp_throughput_u"$UPDATE"_overhead_"$expLabel"/n"$numNodes".txt
    elif [[ $line =~ $PERF_REGEX ]]
    then
        case "${BASH_REMATCH[2]}" in
            cache-misses)
                CACHE_MISS_RATE=$(echo "${BASH_REMATCH[1]} $NUM_OPS" | awk '{printf "%.4f", $1 / $2}')
                echo "$DATA_STRUCTURE $NUM_THREADS $CACHE_MISS_RATE" >> /root/sifter/ASCYLIB/exp_cache-miss-rate_u"$UPDATE"_overhead_"$expLabel"/n"$numNodes".txt
            ;;
            L1-dcache-load-misses)
                L1_DCACHE_LOAD_MISS_RATE=$(echo "${BASH_REMATCH[1]} $NUM_OPS" | awk '{printf "%.4f", $1 / $2}')
                echo "$DATA_STRUCTURE $NUM_THREADS $L1_DCACHE_LOAD_MISS_RATE" >> /root/sifter/ASCYLIB/exp_L1-dcache-load-miss-rate_u"$UPDATE"_overhead_"$expLabel"/n"$numNodes".txt
            ;;
            LLC-load-misses)
                LLC_LOAD_MISS_RATE=$(echo "${BASH_REMATCH[1]} $NUM_OPS" | awk '{printf "%.4f", $1 / $2}')
                echo "$DATA_STRUCTURE $NUM_THREADS $LLC_LOAD_MISS_RATE" >> /root/sifter/ASCYLIB/exp_LLC-load-miss-rate_u"$UPDATE"_overhead_"$expLabel"/n"$numNodes".txt
            ;;
            LLC-store-misses)
                LLC_STORE_MISS_RATE=$(echo "${BASH_REMATCH[1]} $NUM_OPS" | awk '{printf "%.4f", $1 / $2}')
                echo "$DATA_STRUCTURE $NUM_THREADS $LLC_STORE_MISS_RATE" >> /root/sifter/ASCYLIB/exp_LLC-store-miss-rate_u"$UPDATE"_overhead_"$expLabel"/n"$numNodes".txt
            ;;
            page-faults)
                echo "$DATA_STRUCTURE $NUM_THREADS ${BASH_REMATCH[1]}" >> /root/sifter/ASCYLIB/exp_page-faults_u"$UPDATE"_overhead_"$expLabel"/n"$numNodes".txt
            ;;
            context-switches)
                CONTEXT_SWITCH_RATE=$(echo "${BASH_REMATCH[1]} $NUM_OPS" | awk '{printf "%.4f", $1 / $2}')
                echo "$DATA_STRUCTURE $NUM_THREADS $CONTEXT_SWITCH_RATE" >> /root/sifter/ASCYLIB/exp_context-switch-rate_u"$UPDATE"_overhead_"$expLabel"/n"$numNodes".txt
            ;;
            dTLB-load-misses)
                TLB_LOAD_MISS_RATE=$(echo "${BASH_REMATCH[1]} $NUM_OPS" | awk '{printf "%.4f", $1 / $2}')
                echo "$DATA_STRUCTURE $NUM_THREADS $TLB_LOAD_MISS_RATE" >> /root/sifter/ASCYLIB/exp_dTLB-load-miss-rate_u"$UPDATE"_overhead_"$expLabel"/n"$numNodes".txt
            ;;
        esac
    fi
}

for DATA_STRUCTURE in "${dataStructures[@]}"
do
    case $DATA_STRUCTURE in
        ellen)
            preload="./external/lib/libssmem_x86_64.so"
            binName="./bin/lf-bst_ellen"
            make bst_ellen INIT=all SET_CPU=0
        ;;
        drachsler)
            preload="./external/lib/libssmem_x86_64.so"
            binName="./bin/lb-bst-drachsler"
            make bst_drachsler INIT=all SET_CPU=0
        ;;
        bronson)
            preload="./external/lib/libssmem_x86_64.so"
            binName="./bin/lb-bst_bronson"
            make bst_bronson INIT=all SET_CPU=0
        ;;
        ellen-overhead)
            cd /root/sifter/ASCYLIB-ellen
            preload="/root/sifter/memhook/libmemhook.so:./external/lib/libssmem_x86_64.so"
            binName="./bin/lf-bst_ellen"
            make bst_ellen USE_MEMHOOK=1 INIT=all SET_CPU=0
        ;;
        drachsler-overhead)
            cd /root/sifter/ASCYLIB-drachsler
            preload="/root/sifter/memhook/libmemhook.so:./external/lib/libssmem_x86_64.so"
            binName="./bin/lb-bst-drachsler"
            make bst_drachsler USE_MEMHOOK=1 INIT=all SET_CPU=0
        ;;
        bronson-overhead)
            cd /root/sifter/ASCYLIB-bronson
            preload="/root/sifter/memhook/libmemhook.so:./external/lib/libssmem_x86_64.so"
            binName="./bin/lb-bst_bronson"
            make bst_bronson USE_MEMHOOK=1 INIT=all SET_CPU=0
        ;;
    esac

    echo "Starting experiment: $DATA_STRUCTURE"
    for numNodes in 10000000
    do
        echo "  Update percentage: $UPDATE"
        for RUN in {0..9}
        do
            echo "    Run: $RUN"
            for NUM_THREADS in 8 24 96
            do
                echo "      Number of threads: $NUM_THREADS"
                case $NUM_THREADS in
                    4)
                        LD_PRELOAD=$preload numactl -i 0 taskset -c 0-3 perf stat -e cache-misses,page-faults,L1-dcache-load-misses,LLC-load-misses,LLC-store-misses,context-switches,dTLB-load-misses $binName -d 3000 -i $numNodes -n $NUM_THREADS --update-rate $UPDATE 2>&1 | while read line; do
                            processLine "$line"
                        done
                    ;;
                    8)
                        LD_PRELOAD=$preload numactl -i 0 taskset -c 0-7 perf stat -e cache-misses,page-faults,L1-dcache-load-misses,LLC-load-misses,LLC-store-misses,context-switches,dTLB-load-misses $binName -d 3000 -i $numNodes -n $NUM_THREADS --update-rate $UPDATE 2>&1 | while read line; do
                            processLine "$line"
                        done
                    ;;
                    24)
                        LD_PRELOAD=$preload numactl -i 0 taskset -c 0-23 perf stat -e cache-misses,page-faults,L1-dcache-load-misses,LLC-load-misses,LLC-store-misses,context-switches,dTLB-load-misses $binName -d 3000 -i $numNodes -n $NUM_THREADS --update-rate $UPDATE 2>&1 | while read line; do
                            processLine "$line"
                        done
                    ;;
                    96)
                        LD_PRELOAD=$preload numactl -i all taskset -c 0-95 perf stat -e cache-misses,page-faults,L1-dcache-load-misses,LLC-load-misses,LLC-store-misses,context-switches,dTLB-load-misses $binName -d 3000 -i $numNodes -n $NUM_THREADS --update-rate $UPDATE 2>&1 | while read line; do
                            processLine "$line"
                        done
                    ;;
                esac
                rm -f binary_dump.txt
            done
        done
    done
done
